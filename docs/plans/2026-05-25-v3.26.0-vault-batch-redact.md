**Goal:** Ship "Vault" — batch auto-redact across an entire folder of PDFs in one
click, using Slab's existing regex presets (`email`, `ssn`, `phone`, `cc`),
custom patterns, AND an optional offline LLM (Ollama) pass for prose PII —
producing one redacted output per input plus a per-file PII audit CSV.

**Architecture:** A new `pdf::redact_batch` module composes the existing
`pdf::auto_redact::auto_redact()` primitive over a directory walk. CSV
`_slab_vault_audit.csv` rows record per-input: filename, regex_matches,
ai_matches, output_path, sha256(output), status. Tauri command
`slab_vault_batch_redact` streams progress via existing
`emit_batch_progress` channel. Front-end `VaultBatchPanel.svelte` mirrors
`QuillBatchPanel` UX: drop a folder → preview row 1 → run full batch → done.

**Tech Stack:** Rust (`walkdir`, `csv`, `sha2`, `regex`, `lopdf`), Tauri 2,
Svelte 5. All Slab internals already in tree.

```rust
//! `pdf::auto_redact::auto_redact()` with the configured presets + custom

//! workflows require a tamper-evident log of what was scrubbed.

use serde::{Deserialize, Serialize};
use std::path::PathBuf;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BatchRedactSpec {
    pub input_dir: PathBuf,
    pub out_dir: PathBuf,
    #[serde(default)]
    pub patterns: Vec<String>,
    #[serde(default)]
    pub presets: Vec<String>,
    #[serde(default)]

    #[serde(default)]
    pub recurse: bool,
    #[serde(default)]
    pub zip_as: Option<String>,
    /// Optional Ollama provider name; when `Some`, run AI PII pass.
    #[serde(default)]
    pub ai_provider: Option<String>,
    /// Preview mode: only process the first matching PDF.
    #[serde(default)]
    pub only_first: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FileResult {
    pub input: PathBuf,
    pub output: Option<PathBuf>,
    pub regex_matches: u32,
    pub ai_matches: u32,
    pub sha256: Option<String>,
    pub error: Option<String>,
}

    pub files: Vec<FileResult>,

    pub zip_path: Option<PathBuf>,

pub fn run_batch(_spec: BatchRedactSpec) -> Result<BatchRedactReport, crate::pdf::PdfError> {
```

```rust
use std::path::Path;
use walkdir::WalkDir;

pub fn enumerate_pdfs(dir: &Path, recurse: bool) -> Result<Vec<PathBuf>, crate::pdf::PdfError> {
    let max_depth = if recurse { usize::MAX } else { 1 };
    let mut out = Vec::new();
    for entry in WalkDir::new(dir).max_depth(max_depth).follow_links(false) {
        let entry = match entry {
            Ok(e) => e,
            Err(_) => continue,
        };
        if !entry.file_type().is_file() {
            continue;
        }
        let name = entry.file_name().to_string_lossy();
        if name.starts_with('.') {
            continue;
        }
        if name.to_lowercase().ends_with(".pdf") {
            out.push(entry.path().to_path_buf());
        }
    }
    out.sort();
    Ok(out)
}
```

```rust
use crate::pdf::auto_redact::{auto_redact, AutoRedactOpts};

pub fn run_batch(spec: BatchRedactSpec) -> Result<BatchRedactReport, crate::pdf::PdfError> {
    std::fs::create_dir_all(&spec.out_dir).map_err(|e| {
        crate::pdf::PdfError::Other(format!("create out_dir: {e}"))
    })?;

    let mut files = enumerate_pdfs(&spec.input_dir, spec.recurse)?;
    if spec.only_first {
        files.truncate(1);
    }

    let mut report = BatchRedactReport {
        files_total: files.len(),
        audit_csv: spec.out_dir.join("_slab_vault_audit.csv"),
        ..Default::default()
    };

    for path in files {
        let stem = path
            .file_stem()
            .and_then(|s| s.to_str())
            .unwrap_or("doc")
            .to_string();
        let out_path = spec.out_dir.join(format!("{stem}_redacted.pdf"));
        let opts = AutoRedactOpts {
            patterns: spec.patterns.clone(),
            presets: spec.presets.clone(),
            gray: spec.gray,
        };
        let mut fr = FileResult {
            input: path.clone(),
            output: None,
            regex_matches: 0,
            ai_matches: 0,
            sha256: None,
            error: None,
        };
        match auto_redact(&path, &out_path, opts) {
            Ok(hits) => {
                fr.regex_matches = hits;
                fr.output = Some(out_path);
                report.files_succeeded += 1;
                report.total_hits += hits as u64;
            }
            Err(e) => {
                fr.error = Some(format!("{e}"));
                report.files_failed += 1;
            }
        }
        report.files.push(fr);
    }

    Ok(report)
}
```

```rust
    use sha2::{Digest, Sha256};
    use std::io::Read;

    for fr in report.files.iter_mut() {
        if let Some(out) = &fr.output {
            if let Ok(mut f) = std::fs::File::open(out) {
                let mut hasher = Sha256::new();
                let mut buf = [0u8; 8192];
                while let Ok(n) = f.read(&mut buf) {
                    if n == 0 { break; }
                    hasher.update(&buf[..n]);
                }
                fr.sha256 = Some(format!("{:x}", hasher.finalize()));
            }
        }
    }

    let mut wtr = csv::Writer::from_path(&report.audit_csv)
        .map_err(|e| crate::pdf::PdfError::Other(format!("audit csv: {e}")))?;
    wtr.write_record(["input", "output", "regex_matches", "ai_matches",
                      "sha256", "status", "error"])
        .map_err(|e| crate::pdf::PdfError::Other(format!("audit csv: {e}")))?;
    for fr in &report.files {
        let status = if fr.error.is_none() { "OK" } else { "FAILED" };
        wtr.write_record([
            fr.input.display().to_string(),
            fr.output.as_ref().map(|p| p.display().to_string()).unwrap_or_default(),
```

```svelte
<!-- src/lib/components/VaultBatchPanel.svelte -->
<script lang="ts">
  import { invoke } from '@tauri-apps/api/core';
  import { open } from '@tauri-apps/plugin-dialog';
  import { t } from '$lib/i18n';

  let inputDir = $state<string | null>(null);
  let outDir = $state<string | null>(null);
  let presets = $state({ email: true, ssn: true, phone: false, cc: false });
  let recurse = $state(false);
  let zipAs = $state('');
  let running = $state(false);
  let report = $state<any | null>(null);
  let error = $state<string | null>(null);

  async function pickInput() {
    const d = await open({ directory: true, multiple: false });
    if (typeof d === 'string') inputDir = d;
  }
  async function pickOutput() {
    const d = await open({ directory: true, multiple: false });
    if (typeof d === 'string') outDir = d;
  }

  async function run() {
    if (!inputDir || !outDir) {
      error = $t('vault.error.missing_dirs');
      return;
    }
    error = null;
    running = true;
    report = null;
    const selected = Object.entries(presets).filter(([_, v]) => v).map(([k]) => k);
    try {
      report = await invoke('slab_vault_batch_redact', {
        spec: {
          input_dir: inputDir,
          out_dir: outDir,
          patterns: [],
          presets: selected,
          gray: 0.0,
          recurse,
          zip_as: zipAs.trim() === '' ? null : zipAs.trim(),
          ai_provider: null,
          only_first: false,
        },
      });
    } catch (e) {
      error = String(e);
    } finally {
      running = false;
    }
  }
</script>

<section class="vault-panel">
  <header>
    <h2>{$t('vault.title')}</h2>
    <p class="subtitle">{$t('vault.subtitle')}</p>
  </header>

  <div class="row">
    <button onclick={pickInput} disabled={running}>
      {inputDir ? `📁 ${inputDir}` : $t('vault.pick_input')}
    </button>
    <button onclick={pickOutput} disabled={running}>
      {outDir ? `📂 ${outDir}` : $t('vault.pick_output')}
    </button>
  </div>

  <fieldset class="presets" disabled={running}>
    <legend>{$t('vault.presets')}</legend>
    <label><input type="checkbox" bind:checked={presets.email}> Email</label>
    <label><input type="checkbox" bind:checked={presets.ssn}> SSN</label>
    <label><input type="checkbox" bind:checked={presets.phone}> Phone</label>
    <label><input type="checkbox" bind:checked={presets.cc}> Credit card</label>
  </fieldset>

  <label class="recurse"><input type="checkbox" bind:checked={recurse} disabled={running}>
    {$t('vault.recurse')}</label>

  <label class="zip">
    {$t('vault.zip_as')}
    <input type="text" bind:value={zipAs} placeholder="vault" disabled={running}>
  </label>

  <button class="run" onclick={run} disabled={running || !inputDir || !outDir}>
    {running ? $t('vault.running') : $t('vault.run')}
  </button>

  {#if error}
    <div class="error">{error}</div>
  {/if}

  {#if report}
    <div class="report">
      <div class="big">{report.files_succeeded}/{report.files_total} {$t('vault.files_done')}</div>
      <div class="mid">{report.total_hits} {$t('vault.hits')}</div>

      </div>
    </div>
  {/if}
</section>
```
