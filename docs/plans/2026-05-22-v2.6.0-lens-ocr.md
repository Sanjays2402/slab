**Step 2: Implement `parse_hocr_words`**

Use a minimal regex parser (`regex` already in workspace deps) to walk `class="ocrx_word"` spans, pull `bbox` and `x_wconf` from the `title` attribute, convert pixel coordinates → PDF points using `pts = (px / dpi) * 72`, and emit `WordBox`.

```rust
use regex::Regex;

pub fn parse_hocr_words(
    hocr: &str,
    page: u32,
    page_width_pts: f32,
    page_height_pts: f32,
    dpi: u32,
) -> Result<Vec<WordBox>, PdfError> {
    let re = Regex::new(
        r#"<span[^>]*class=['"]ocrx_word['"][^>]*title=['"][^'"]*bbox (\d+) (\d+) (\d+) (\d+);[^'"]*x_wconf (\d+)[^'"]*['"][^>]*>([^<]+)</span>"#,
    ).map_err(|e| PdfError::Other(format!("hocr regex: {e}")))?;
    let scale = 72.0 / dpi as f32;
    let mut out = Vec::new();
    for cap in re.captures_iter(hocr) {
        let x0: f32 = cap[1].parse().unwrap_or(0.0);
        let y0: f32 = cap[2].parse().unwrap_or(0.0);
        let x1: f32 = cap[3].parse().unwrap_or(0.0);
        let y1: f32 = cap[4].parse().unwrap_or(0.0);
        let conf: f32 = cap[5].parse().unwrap_or(0.0);
        let text = cap[6].to_string();
        // HOCR origin is top-left; PDF origin is bottom-left. Flip Y.
        let y_min_pts = page_height_pts - y1 * scale;
        let y_max_pts = page_height_pts - y0 * scale;
        out.push(WordBox {
            page,
            text,
            bbox_pts: [x0 * scale, y_min_pts, x1 * scale, y_max_pts],
            confidence: conf,
        });
    }
    Ok(out)
}
```

```
({escaped_word}) Tj
```

```rust
            confidence: info.confidence() as f32,
```

```rust
#[derive(Debug, Clone, Serialize)]
pub enum BatchEventKind { FileStarted, FileCompleted, FileFailed, BatchDone }

#[derive(Debug, Clone, Serialize)]
pub struct BatchEvent {
    pub kind: BatchEventKind,
    pub index: u32,
    pub total: u32,
    pub path: String,
    pub error: Option<String>,
    pub words: u32,
    pub mean_confidence: f32,
}

pub fn batch_ocr_folder<F: FnMut(BatchEvent)>(
    input_dir: &Path,
    output_dir: PathBuf,
    opts: &OcrV2Opts,
    mut on_event: F,
) -> Result<BatchSummary, PdfError> {
    std::fs::create_dir_all(&output_dir).ok();
    let pdfs: Vec<_> = std::fs::read_dir(input_dir)
        .map_err(|e| PdfError::Other(format!("read folder: {e}")))?
        .filter_map(|e| e.ok())
        .filter(|e| e.path().extension().and_then(|s| s.to_str()) == Some("pdf"))
        .map(|e| e.path())
        .collect();
    let total = pdfs.len() as u32;
    let mut succeeded = 0;
    let mut failed = 0;
    for (i, p) in pdfs.iter().enumerate() {
        on_event(BatchEvent {
            kind: BatchEventKind::FileStarted,
            index: i as u32 + 1, total, path: p.to_string_lossy().into(),
            error: None, words: 0, mean_confidence: 0.0,
        });
        let stem = p.file_stem().and_then(|s| s.to_str()).unwrap_or("doc");
        let out = output_dir.join(format!("{stem}-ocr.pdf"));
        match ocr_with_text_layer(p, &out, opts) {
            Ok(r) => {
                succeeded += 1;
                on_event(BatchEvent {
                    kind: BatchEventKind::FileCompleted,
                    index: i as u32 + 1, total, path: p.to_string_lossy().into(),
                    error: None, words: r.words_recognised, mean_confidence: r.mean_confidence,
                });
            }
            Err(e) => {
                failed += 1;
                on_event(BatchEvent {
                    kind: BatchEventKind::FileFailed,
                    index: i as u32 + 1, total, path: p.to_string_lossy().into(),
                    error: Some(e.to_string()), words: 0, mean_confidence: 0.0,
                });
            }
        }
    }
    on_event(BatchEvent {
        kind: BatchEventKind::BatchDone,
        index: total, total, path: input_dir.to_string_lossy().into(),
        error: None, words: 0, mean_confidence: 0.0,
    });
    Ok(BatchSummary { total, succeeded, failed })
}

#[derive(Debug, Clone, Serialize)]
pub struct BatchSummary { pub total: u32, pub succeeded: u32, pub failed: u32 }
```

```rust
use crate::pdf::PdfError;
use crate::pdf::table_extract::Table;
use rust_xlsxwriter::{Workbook, Format, FormatBorder};
use serde::Serialize;
use std::path::Path;

#[derive(Debug, Clone, Serialize)]
pub struct XlsxReport {
    pub sheets_written: u32,
    pub rows_total: u32,
    pub cells_total: u32,
}

pub fn export_tables_to_xlsx(tables: &[Table], out: &Path) -> Result<XlsxReport, PdfError> {
    if tables.is_empty() {
        return Err(PdfError::Other("no tables to export".into()));
    }
    let mut wb = Workbook::new();
    let header_fmt = Format::new().set_bold().set_background_color("#F0E9D2").set_border(FormatBorder::Thin);
    let cell_fmt = Format::new().set_border(FormatBorder::Hair);
    let mut rows_total = 0u32;
    let mut cells_total = 0u32;
    for t in tables {
        let sheet = wb.add_worksheet();
        let name = format!("Page{}_Table{}", t.page, t.index);
        sheet.set_name(&name[..name.len().min(31)])
            .map_err(|e| PdfError::Other(format!("sheet name: {e}")))?;
        // Freeze top row
        sheet.set_freeze_panes(1, 0).ok();
        for (r, row) in t.rows.iter().enumerate() {
            for (c, val) in row.iter().enumerate() {
                let fmt = if r == 0 { &header_fmt } else { &cell_fmt };
                sheet.write_string_with_format(r as u32, c as u16, val, fmt)
                    .map_err(|e| PdfError::Other(format!("write cell: {e}")))?;
                cells_total += 1;
            }
            rows_total += 1;
        }
        // Autofit columns
        sheet.autofit();
    }
    wb.save(out).map_err(|e| PdfError::Other(format!("xlsx save: {e}")))?;
    Ok(XlsxReport {
        sheets_written: tables.len() as u32,
        rows_total,
        cells_total,
    })
}
```
