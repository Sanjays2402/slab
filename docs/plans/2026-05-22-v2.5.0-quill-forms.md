**Goal:** Open any PDF with form fields (W-9, I-9, tax form, court filing, insurance claim, vendor onboarding) → fill every field inline in the Slab reader → save a filled PDF that opens identically in Adobe — entirely offline.

```rust
//! AcroForm fill & flatten (v2.5.0 "Quill").
//!
//! Resolves PDF /AcroForm fields, fills user values, optionally bakes the
//! field appearance permanently into page content streams ("flatten").

pub mod fdf;
pub mod fill;
pub mod flatten;
pub mod inspect;

use serde::{Deserialize, Serialize};

/// One interactive form field.
#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct FormField {
    /// Fully-qualified name (e.g. `"f1_01[0]"` or `"address.street"`).
    pub name: String,
    /// Field kind.
    pub kind: FieldKind,
    /// 0-based page index where this field's widget annotation lives.
    pub page_index: u32,
    /// Widget rect on the page in PDF user space `[x0 y0 x1 y1]`.
    pub rect: [f32; 4],
    /// Current value (None = unset).
    pub value: Option<String>,
    /// Allowed values for radio groups / dropdowns.
    pub options: Vec<String>,
    /// True if the field has `/Ff` ReadOnly bit set.
    pub read_only: bool,
    /// True if the field is required (`/Ff` bit 2).
    pub required: bool,
    /// True if multi-line text field (`/Ff` bit 13 for Tx).
    pub multiline: bool,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
#[serde(tag = "kind", rename_all = "lowercase")]
pub enum FieldKind {
    Text,
    Checkbox,
    Radio,
    Choice, // /Ch — combo or list
    Signature, // /Sig
    Button,   // pushbutton — non-data
}

/// Top-level inspect report.
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FormReport {
    pub has_acroform: bool,
    pub fields: Vec<FormField>,
    /// Tab order indices into `fields`, derived from /AcroForm /Fields order
    /// (NOT page /Annots — the spec mandates that for forms).
    pub tab_order: Vec<u32>,
}

/// Errors specific to AcroForm operations.
#[derive(Debug, thiserror::Error)]
pub enum FormError {
    #[error("PDF has no /AcroForm dictionary")]
    NoAcroForm,
    #[error("Field {0:?} not found")]
    UnknownField(String),
    #[error("Field {0:?} is read-only")]
    ReadOnly(String),
    #[error("Value {value:?} not in allowed options for field {field:?}")]
    InvalidOption { field: String, value: String },
    #[error(transparent)]
    Pdf(#[from] crate::pdf::PdfError),
    #[error(transparent)]
    Lopdf(#[from] lopdf::Error),
    #[error(transparent)]
    Io(#[from] std::io::Error),
}
```

### Step 3: Implement `pub fn inspect(path: &Path) -> Result<FormReport, FormError>`

Skeleton:
```rust
use lopdf::{Dictionary, Document, Object, ObjectId};
use std::path::Path;
use super::*;

pub fn inspect(path: &Path) -> Result<FormReport, FormError> {
    let doc = Document::load(path).map_err(FormError::Lopdf)?;
    let catalog = doc.catalog().map_err(FormError::Lopdf)?;
    let Some(acroform_obj) = catalog.get(b"AcroForm").ok() else {
        return Ok(FormReport { has_acroform: false, fields: vec![], tab_order: vec![] });
    };
    let acroform = resolve_dict(&doc, acroform_obj)?;
    let Some(fields_arr) = acroform.get(b"Fields").ok().and_then(|o| o.as_array().ok()) else {
        return Ok(FormReport { has_acroform: true, fields: vec![], tab_order: vec![] });
    };

    // Build page-id → page-index map (one pass over /Pages).
    let page_index_of: std::collections::HashMap<ObjectId, u32> =
        doc.get_pages().iter().enumerate()
            .map(|(i, (_n, id))| (*id, i as u32))
            .collect();

    let mut out = Vec::new();
    for obj in fields_arr {
        walk_field(&doc, obj, &page_index_of, None, &mut out)?;
    }

    let tab_order = (0..out.len() as u32).collect();
    Ok(FormReport { has_acroform: true, fields: out, tab_order })
}

fn walk_field(
    doc: &Document,
    obj: &Object,
    page_idx: &std::collections::HashMap<ObjectId, u32>,
    inherited_parent_name: Option<&str>,
    out: &mut Vec<FormField>,
) -> Result<(), FormError> {
    let dict = resolve_dict(doc, obj)?;
    // /T = partial field name; concatenate with parent via "."
    let partial = dict.get(b"T").ok()
        .and_then(|o| o.as_str().ok())
        .map(|b| String::from_utf8_lossy(b).into_owned())
        .unwrap_or_default();
    let full_name = match inherited_parent_name {
        Some(p) if !partial.is_empty() => format!("{}.{}", p, partial),
        Some(p) => p.to_string(),
        None => partial.clone(),
    };

    // Children = /Kids that are *fields* (have /T). Widgets are /Kids w/o /T.
    if let Ok(kids) = dict.get(b"Kids").and_then(|o| o.as_array()) {
        // Distinguish: if any kid has /T, this is a field group → recurse.
        // Otherwise the /Kids are widget annotations for *this* field.
        let mut any_field_kid = false;
        for kid in kids {
            if let Ok(kd) = resolve_dict(doc, kid) {
                if kd.get(b"T").is_ok() {
                    any_field_kid = true;
                    break;
                }
            }
        }
        if any_field_kid {
            for kid in kids {
                walk_field(doc, kid, page_idx, Some(&full_name), out)?;
            }
            return Ok(());
        }
    }

    // Leaf field. Determine kind from /FT (inherited).
    let ft = inherited_ft(doc, &dict, obj)?;
    let ff = dict.get(b"Ff").ok().and_then(|o| o.as_i64().ok()).unwrap_or(0);
    let kind = classify(&ft, ff);

    // Rect: from widget. If single-widget field, /Rect lives on this dict.
    // Multi-widget field (e.g. one checkbox group) — emit one entry per /Kid widget.
    let (rect, page_index) = first_widget_rect(doc, &dict, page_idx)?;

    let value = dict.get(b"V").ok().map(|v| stringify_value(doc, v));
    let options = extract_options(doc, &dict);

    out.push(FormField {
        name: full_name,
        kind,
        page_index,
        rect,
        value,
        options,
        read_only: ff & (1 << 0) != 0,
        required: ff & (1 << 1) != 0,
        multiline: matches!(ft.as_str(), "Tx") && (ff & (1 << 12)) != 0,
    });
    Ok(())
}

// — Helpers (resolve_dict, inherited_ft, classify, first_widget_rect,
//    stringify_value, extract_options) follow standard lopdf idioms.
//    See PDF 32000-1 §12.7.4 for `/FT` values: Btn, Tx, Ch, Sig.
//    For /Btn /Ff bit 16 (Radio) and bit 17 (PushButton) distinguish kinds.
```

### Step 2: Implement `pub fn fill(...)`

```rust
use lopdf::{Document, Object, ObjectId, dictionary};
use serde::{Deserialize, Serialize};
use std::collections::HashMap;
use std::path::Path;
use super::*;

#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct FillOpts {
    /// If true, set `/AcroForm /NeedAppearances true` so viewers regenerate
    /// the field appearance streams on next open. Default: true.
    #[serde(default = "default_need_appearances")]
    pub need_appearances: bool,
    /// If true, allow Yes/No-style booleans like "true", "1", "x" to set
    /// a checkbox to its first non-Off option. Default: true.
    #[serde(default = "default_bool_coerce")]
    pub bool_coerce: bool,
}
fn default_need_appearances() -> bool { true }
fn default_bool_coerce() -> bool { true }

pub fn fill(
    src: &Path,
    dst: &Path,
    values: &HashMap<String, String>,
    opts: &FillOpts,
) -> Result<FillReport, FormError> {
    let mut doc = Document::load(src)?;
    let report = super::inspect::inspect(src)?;

    // Build name → ObjectId map by re-walking. (Could be merged with inspect()
    // later for perf; keeping separate keeps inspect read-only.)
    let id_by_name = super::inspect::field_object_ids(&doc)?;

    let mut applied = 0usize;
    for (name, raw) in values {
        let field = report.fields.iter().find(|f| &f.name == name)
            .ok_or_else(|| FormError::UnknownField(name.clone()))?;
        if field.read_only {
            return Err(FormError::ReadOnly(name.clone()));
        }
        let id = id_by_name.get(name).copied()
            .ok_or_else(|| FormError::UnknownField(name.clone()))?;

        let pdf_value = encode_value(field, raw, opts)?;
        let obj = doc.get_object_mut(id)?;
        if let Object::Dictionary(d) = obj {
            d.set("V", pdf_value);
        }
        applied += 1;
    }

    if opts.need_appearances {
        set_need_appearances(&mut doc)?;
    }

    doc.save(dst)?;
    Ok(FillReport { fields_filled: applied })
}

fn encode_value(field: &FormField, raw: &str, opts: &FillOpts) -> Result<Object, FormError> {
    match field.kind {
        FieldKind::Text | FieldKind::Choice => Ok(Object::string_literal(raw)),
        FieldKind::Checkbox => {
            let on = if opts.bool_coerce {
                matches!(raw.to_lowercase().as_str(), "yes" | "true" | "1" | "x" | "on")
            } else {
                field.options.iter().any(|o| o == raw && o != "Off")
            };
            let name = if on {
                field.options.iter().find(|o| *o != "Off")
                    .cloned()
                    .unwrap_or_else(|| "Yes".to_string())
            } else {
                "Off".to_string()
            };
            Ok(Object::Name(name.into_bytes()))
        }
        FieldKind::Radio => {
            if !field.options.iter().any(|o| o == raw) {
                return Err(FormError::InvalidOption {
                    field: field.name.clone(),
                    value: raw.to_string(),
                });
            }
            Ok(Object::Name(raw.as_bytes().to_vec()))
        }
        FieldKind::Signature | FieldKind::Button => {
            Err(FormError::ReadOnly(field.name.clone()))
        }
    }
}
```

### Step 3: Implement `set_need_appearances` + `field_object_ids`

```rust
fn set_need_appearances(doc: &mut Document) -> Result<(), FormError> {
    let catalog_id = doc.catalog_id().ok_or(FormError::NoAcroForm)?;
    // Resolve AcroForm — may be reference.
    let acroform_id = {
        let catalog = doc.get_object(catalog_id)?.as_dict()?;
        match catalog.get(b"AcroForm")? {
            Object::Reference(r) => Some(*r),
            _ => None,
        }
    };
    if let Some(id) = acroform_id {
        let af = doc.get_object_mut(id)?.as_dict_mut()?;
        af.set("NeedAppearances", true);
    } else {
        // Inline /AcroForm dict in catalog.
        let catalog = doc.get_object_mut(catalog_id)?.as_dict_mut()?;
        if let Ok(Object::Dictionary(d)) = catalog.get_mut(b"AcroForm") {
            d.set("NeedAppearances", true);
        }
    }
    Ok(())
}
```

### Step 2: Implement `flatten`

```rust
pub fn flatten(src: &Path, dst: &Path) -> Result<FlattenReport, FormError> {
    let mut doc = Document::load(src)?;

    // 1. Walk pages, drop every /Annot of /Subtype /Widget. Append a small
    //    content stream chunk per page that BT/Tf/Tj-emits each widget's
    //    /V text at the widget's /Rect origin using Helvetica @ 10pt.
    let pages: Vec<(u32, ObjectId)> = doc.get_pages().into_iter().collect();
    let report = super::inspect::inspect(src)?;

    let mut pages_touched = 0u32;
    for (page_num, page_id) in pages {
        let page_widgets: Vec<&FormField> = report.fields.iter()
            .filter(|f| f.page_index == page_num - 1)
            .collect();
        if page_widgets.is_empty() {
            continue;
        }
        let painted = paint_widgets_on_page(&mut doc, page_id, &page_widgets)?;
        if painted > 0 { pages_touched += 1; }
        strip_widget_annots(&mut doc, page_id)?;
    }

    // 2. Remove /AcroForm from catalog.
    remove_acroform(&mut doc)?;

    doc.save(dst)?;
    Ok(FlattenReport { pages_touched, fields_baked: report.fields.len() as u32 })
}
```

`paint_widgets_on_page` writes a new content stream chunk:

```
q
BT
/Helv 10 Tf
0 0 0 rg
<x> <y> Td
(<escaped value>) Tj
ET
Q
```

…appended to (or merged with) the page's existing `/Contents` stream. Use the
existing `crate::pdf::content_stream::append_chunk()` helper (see `header_footer.rs`
for prior art that already paints text on every page). Resource the
`/Helvetica` Type1 font via the page's `/Resources /Font` entry, adding it
under name `Helv` if missing.

For now (v2.5.0) checkboxes flatten as their /V name in 8pt monospace —
follow-up tick can render real check glyphs.

```ts
import { invoke } from '@tauri-apps/api/core';

export type FieldKind =
  | 'text' | 'checkbox' | 'radio' | 'choice' | 'signature' | 'button';

export interface FormField {
  name: string;
  kind: FieldKind;
  page_index: number;
  rect: [number, number, number, number];
  value: string | null;
  options: string[];
  read_only: boolean;
  required: boolean;
  multiline: boolean;
}

export interface FormReport {
  has_acroform: boolean;
  fields: FormField[];
  tab_order: number[];
}

export const formInspect = (path: string) =>
  invoke<FormReport>('slab_form_inspect', { path });

export const formFill = (
  src: string,
  dst: string,
  values: Record<string, string>,
  opts?: { need_appearances?: boolean; bool_coerce?: boolean },
) => invoke<{ fields_filled: number }>('slab_form_fill', { src, dst, values, opts });

export const formFlatten = (src: string, dst: string) =>
  invoke<{ pages_touched: number; fields_baked: number }>(
    'slab_form_flatten', { src, dst });

export const formExportFdf = (src: string, dst: string) =>
  invoke<number>('slab_form_export_fdf', { src, dst });

export const formImportFdf = (src: string, fdf: string, dst: string) =>
  invoke<{ fields_filled: number }>('slab_form_import_fdf', { src, fdf, dst });
```

## Slice 5: FormPanel Sidebar — 75 min

**Objective:** Sidebar panel that lists every detected field, lets users type a value, and saves either as a filled PDF or a flattened PDF.

**File:** Create `src/lib/panels/FormPanel.svelte` (~280 LOC).

Skeleton:
```svelte
<script lang="ts">
  import { formInspect, formFill, formFlatten, type FormField, type FormReport } from '$lib/api/forms';
  import { currentTab } from '$lib/stores/tabs';
  import { save as saveDialog } from '@tauri-apps/plugin-dialog';
  import { t } from '$lib/i18n';

  let report = $state<FormReport | null>(null);
  let values = $state<Record<string, string>>({});
  let flattenOnSave = $state(true);
  let loading = $state(false);
  let error = $state<string | null>(null);

  $effect(() => {
    const path = $currentTab?.path;
    if (!path) { report = null; return; }
    loading = true;
    formInspect(path)
      .then(r => { report = r; values = Object.fromEntries(r.fields.map(f => [f.name, f.value ?? ''])); })
      .catch(e => { error = String(e); })
      .finally(() => { loading = false; });
  });

  async function save() {
    if (!report || !$currentTab) return;
    const dst = await saveDialog({ defaultPath: $currentTab.path.replace(/\.pdf$/, '-filled.pdf') });
    if (!dst) return;
    loading = true;
    try {
      const tmp = `${dst}.tmp`;
      await formFill($currentTab.path, tmp, values, { need_appearances: true });
      if (flattenOnSave) {
        await formFlatten(tmp, dst);
      } else {
        // Rename tmp → dst via Tauri fs plugin.
      }
      // Trigger WOW ink-settle animation.
      window.dispatchEvent(new CustomEvent('slab:forms-settle', { detail: { count: report.fields.length } }));
    } catch (e) { error = String(e); }
    finally { loading = false; }
  }
</script>

<div class="form-panel">
  <header>
    <h2>{$t('forms.title')}</h2>
    {#if report?.has_acroform}
      <span class="badge">{report.fields.length} {$t('forms.fields_detected')}</span>
    {/if}
  </header>

  {#if loading}
    <div class="skeleton">...</div>
  {:else if !report}
    <p class="empty">{$t('forms.empty.no_doc')}</p>
  {:else if !report.has_acroform}
    <p class="empty designed">
      <svg>...</svg>
      {$t('forms.empty.no_form')}
    </p>
  {:else}
    <ol class="field-list">
      {#each report.fields as field, i}
        <li>
          <label>
            <span class="num">{i + 1}</span>
            <span class="name">{field.name}</span>
            {#if field.required}<span class="req">*</span>{/if}
          </label>
          {#if field.kind === 'text'}
            {#if field.multiline}
              <textarea bind:value={values[field.name]} rows="3" />
            {:else}
              <input type="text" bind:value={values[field.name]} />
            {/if}
          {:else if field.kind === 'checkbox'}
            <input type="checkbox"
                   checked={values[field.name] !== 'Off' && values[field.name] !== ''}
                   on:change={(e) => values[field.name] = (e.currentTarget as HTMLInputElement).checked ? (field.options.find(o => o !== 'Off') ?? 'Yes') : 'Off'} />
          {:else if field.kind === 'radio'}
            {#each field.options as opt}
              <label class="radio">
                <input type="radio" name={field.name} value={opt}
                       checked={values[field.name] === opt}
                       on:change={() => values[field.name] = opt} />
                {opt}
              </label>
            {/each}
          {:else if field.kind === 'choice'}
            <select bind:value={values[field.name]}>
              {#each field.options as opt}<option>{opt}</option>{/each}
            </select>
          {:else}
            <em>({field.kind} — read-only)</em>
          {/if}
        </li>
      {/each}
    </ol>

    <footer>
      <label class="flatten-toggle">
        <input type="checkbox" bind:checked={flattenOnSave} />
        {$t('forms.flatten_on_save')}
      </label>
      <button class="primary" on:click={save} disabled={loading}>
        {$t('forms.save')}
      </button>
    </footer>
  {/if}

  {#if error}<p class="error">{error}</p>{/if}
</div>

</style>
```

## Slice 6: Reader Overlay + WOW Ink-Settle Animation — 90 min

**Objective:** When FormPanel is active, overlay native HTML inputs anchored to each widget's `/Rect` directly on top of the pdfjs canvas. Editing a field on the canvas updates `values` two-way. On Save, every input plays the 220ms gold-ink-settle animation.

**File:** Create `src/lib/panels/FormOverlay.svelte`.

Key bits:
- Mount on `.pdfjs-container` for the active page only.
- Position with `transform: matrix(...)` derived from pdfjs's viewport transform (already exposed via `pageView.viewport.transform`).
- `Tab` cycles through `report.tab_order` — `preventDefault` on the global Tab when overlay is active.
- Focus ring = accent gold @ 2px solid + 8px halo with `prefers-reduced-motion` short-circuit.
- WOW: on `slab:forms-settle` event, every input gets class `.settling` applied with `animation-delay: calc(var(--i) * 30ms)`. Keyframes:
  ```css
  @keyframes ink-settle {
    0% { background: rgba(212, 175, 55, 0.18); }
    40% { background: rgba(212, 175, 55, 0.30); transform: translateY(-1px); }
    100% { background: transparent; transform: none; }
  }
  .settling::after {
    /* underline 0→100% gold */
    content: ''; position: absolute; bottom: -2px; left: 0; right: 0;
    height: 2px; background: var(--accent-gold);
    transform-origin: left; transform: scaleX(0);
    animation: settle-underline 220ms cubic-bezier(0.34, 1.56, 0.64, 1) forwards;
  }
  @keyframes settle-underline { to { transform: scaleX(1); } }
  @media (prefers-reduced-motion: reduce) {
    .settling, .settling::after { animation: none !important; }
  }
  ```

## Open Questions (defer to v2.5.1)

1. **FDF import.** Slice 4 ships export only. Import requires an FDF parser
   that handles `/V`, `/T`, `/Kids` recursion — a clean follow-up.
2. **Real checkbox glyphs on flatten.** Right now we paint the `/V` name in
   monospace. Replace with `☑` / `☐` Unicode glyphs once we embed a fallback
   font with those codepoints.
3. **JavaScript actions on field change** (PDF spec §12.6.4 JavaScript
   actions). Slab will not execute embedded JS — security wedge. v2.5.1
   ships a "this form expects JavaScript that won't run" warning banner.
4. **Digital signature capture on `/Sig` fields.** This is the v2.6.0
   "Notary" feature (already roadmapped). v2.5.0 surfaces them as read-only.
5. **Bulk fill from CSV** — drop a 200-row CSV onto a W-9 → get 200 filled
   PDFs. Enterprise wedge. v2.5.1 candidate.

These are explicit follow-ups, not gaps in v2.5.0's promise.
