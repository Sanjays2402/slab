**Goal:** Detect form-field candidates on any **flat** PDF (no AcroForm yet) by analysing page geometry and text — horizontal underlines, square boxes, labeled blanks, and printed checkbox glyphs — and propose a list of `FieldDraft`s the user can accept-all or curate before the Designer commits them.

Re-uses v3.26.0 `forms_design::FieldDraft` as the output shape so the Designer panel accepts the candidates directly: detect → review modal → "Add all" → existing `slab_forms_design_add` writes the AcroForm. Headless-friendly (no pdfjs in Rust) — pure geometry + text on the lopdf side; the UI runs the Designer's existing pdfjs preview for review.

**Tech Stack:** Rust (`lopdf`), Tauri 2, Svelte 5 runes, TypeScript, `pdfjs-dist` (already on disk).

//! Output feeds directly into v3.26.0 Designer (`FieldDraft`) so the user

use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "kind", rename_all = "lowercase")]
pub enum CandidateKind {
    Text { multiline: bool },
    Checkbox,
    Signature,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "type", rename_all = "snake_case")]
pub enum Evidence {
    /// A thin horizontal line with no text on it (an underline-blank).
    HorizontalRule { line_width_pt: f32 },
    /// A small square stroked rectangle (printed checkbox glyph).
    EmptyBox { side_pt: f32 },
    /// A short text run ending with a colon followed by whitespace then a rule.
    LabeledBlank,
    /// A "Signature:" or "Sign here" label followed by a long rule.
    SignatureLine,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct FieldCandidate {
    pub suggested_name: String,
    #[serde(flatten)]
    pub kind: CandidateKind,
    pub page: u32,
    pub rect: [f32; 4],
    pub label: Option<String>,
    pub evidence: Evidence,
    /// 0.0..1.0 — calibrated against the test corpus, surfaced in the UI so
    /// the reviewer can sort low-confidence picks to the top.
    pub confidence: f32,
}

pub struct DetectionReport {
    pub candidates: Vec<FieldCandidate>,
    pub pages_scanned: u32,

    /// committing (we'd merge, not replace, but the user should know).
    pub already_has_acroform: bool,
    /// Soft warnings (e.g. encrypted page, scanned-image-only page).
    pub warnings: Vec<String>,
}

pub(crate) mod geom {
    use lopdf::{content::Content, Document};
    use crate::pdf::PdfError;

    #[derive(Debug, Clone, Copy)]
    pub struct Line { pub x0: f32, pub y0: f32, pub x1: f32, pub y1: f32, pub width: f32 }

    #[derive(Debug, Clone, Copy)]
    pub struct Rect { pub x0: f32, pub y0: f32, pub x1: f32, pub y1: f32, pub stroked: bool }

    #[derive(Debug, Default, Clone)]
    pub struct PagePrims { pub lines: Vec<Line>, pub rects: Vec<Rect> }

    pub fn walk_page(doc: &Document, page_1based: u32) -> Result<PagePrims, PdfError> {
        let page_id = *doc.get_pages().get(&page_1based)
            .ok_or_else(|| PdfError::Other(format!("page {page_1based} missing")))?;
        let data = doc.get_page_content(page_id).map_err(|e| PdfError::Other(e.to_string()))?;
        let content = Content::decode(&data).map_err(|e| PdfError::Other(e.to_string()))?;
        let mut out = PagePrims::default();
        // Graphics state we care about: CTM is identity for v1; line width = 1.
        let mut lw: f32 = 1.0;
        let mut path_start: Option<(f32, f32)> = None;
        let mut path_cur: Option<(f32, f32)> = None;
        let mut pending_rect: Option<(f32, f32, f32, f32)> = None;
        for op in content.operations {
            match op.operator.as_str() {
                "w" => { if let Some(v) = op.operands.first().and_then(num) { lw = v; } }
                "m" => {
                    if let (Some(x), Some(y)) = (op.operands.first().and_then(num),
                                                  op.operands.get(1).and_then(num)) {
                        path_start = Some((x,y)); path_cur = Some((x,y));
                    }
                }
                "l" => {
                    if let (Some(x), Some(y), Some((sx, sy))) =
                        (op.operands.first().and_then(num),
                         op.operands.get(1).and_then(num),
                         path_cur)
                    {
                        // Only stroke happens on 'S' — record provisionally.
                        out.lines.push(Line { x0: sx, y0: sy, x1: x, y1: y, width: lw });
                        path_cur = Some((x,y));
                    }
                }
                "re" => {
                    if let (Some(x), Some(y), Some(w), Some(h)) = (
                        op.operands.first().and_then(num),
                        op.operands.get(1).and_then(num),
                        op.operands.get(2).and_then(num),
                        op.operands.get(3).and_then(num)) {
                        pending_rect = Some((x,y,x+w,y+h));
                    }
                }
                "S" | "s" => {
                    if let Some((x0,y0,x1,y1)) = pending_rect.take() {
                        out.rects.push(Rect { x0,y0,x1,y1, stroked: true });
                    }
                    // lines are already pushed at 'l' time — keep them
                }
                "n" => {
                    // No-paint — drop pending rect AND last unstroked line(s)
                    pending_rect = None;
                    // (For v1 we accept the over-count and filter later by stroke heuristics.)
                }
                _ => {}
            }
        }
        Ok(out)
    }

    fn num(o: &lopdf::Object) -> Option<f32> {
        match o {
            lopdf::Object::Integer(i) => Some(*i as f32),
            lopdf::Object::Real(r) => Some(*r),
            _ => None,
        }
    }
}

pub(crate) mod text {
    use lopdf::{content::Content, Document};
    use crate::pdf::PdfError;

    #[derive(Debug, Clone)]
    pub struct TextRun { pub text: String, pub x: f32, pub y: f32, pub size: f32 }

    pub fn walk_page(doc: &Document, page_1based: u32) -> Result<Vec<TextRun>, PdfError> {
        let page_id = *doc.get_pages().get(&page_1based)
            .ok_or_else(|| PdfError::Other(format!("page {page_1based} missing")))?;
        let data = doc.get_page_content(page_id).map_err(|e| PdfError::Other(e.to_string()))?;
        let content = Content::decode(&data).map_err(|e| PdfError::Other(e.to_string()))?;
        let mut out = Vec::new();
        let mut tx: f32 = 0.0; let mut ty: f32 = 0.0; let mut tf: f32 = 12.0;
        let mut in_text = false;
        for op in content.operations {
            match op.operator.as_str() {
                "BT" => { in_text = true; tx = 0.0; ty = 0.0; }
                "ET" => { in_text = false; }
                "Tf" => {
                    if let Some(s) = op.operands.get(1).and_then(super::geom_num) { tf = s; }
                }
                "Td" | "TD" => {
                    if let (Some(x), Some(y)) = (op.operands.first().and_then(super::geom_num),
                                                 op.operands.get(1).and_then(super::geom_num)) {
                        tx += x; ty += y;
                    }
                }
                "Tj" | "'" => {
                    if in_text {
                        if let Some(lopdf::Object::String(bytes, _)) = op.operands.first() {
                            let s = String::from_utf8_lossy(bytes).into_owned();
                            out.push(TextRun { text: s, x: tx, y: ty, size: tf });
                        }
                    }
                }
                "TJ" => {
                    if in_text {
                        if let Some(lopdf::Object::Array(arr)) = op.operands.first() {
                            let mut s = String::new();
                            for el in arr {
                                if let lopdf::Object::String(b, _) = el {
                                    s.push_str(&String::from_utf8_lossy(b));
                                }
                            }
                            if !s.is_empty() {
                                out.push(TextRun { text: s, x: tx, y: ty, size: tf });
                            }
                        }
                    }
                }
                _ => {}
            }
        }
        Ok(out)
    }
}

### Task 4: Heuristic detector — emit `FieldCandidate`s

**Objective:** Combine `geom::walk_page` + `text::walk_page` into a single `detect_page` function that returns `Vec<FieldCandidate>` with these rules:

1. **HorizontalRule** — A line where `|y1-y0| < 1pt`, `|x1-x0| > 50pt`, `width < 2pt`. Bounding rect = `[x0, y0-2, x1, y0+14]` (14pt of writing room above the rule).
2. **EmptyBox** — A stroked rect where `8 <= side <= 22pt` and `|w-h| < 2pt`. Type = Checkbox. Rect kept as-is.
3. **LabeledBlank** — A `TextRun` ending in `:` (or matching `r"(?i)(name|date|address|signature|email|phone|amount)\s*:?\s*$"`) whose baseline is `<=4pt` above a HorizontalRule starting `<=40pt` to the right of the run's end. Promote the rule to a Text candidate; copy the label (minus trailing `:`); generate a `suggested_name` via slugify; boost confidence to 0.9.
4. **SignatureLine** — Label matches `(?i)(signature|sign here|signed by|x\s*$)` → CandidateKind::Signature, confidence 0.85.

Confidence formula: base 0.55 for raw rules, +0.15 if width/height looks like writing room, +0.20 if a label sits on top of it.

use std::path::Path;
use crate::pdf::PdfError;

pub fn detect(input: &Path) -> Result<DetectionReport, PdfError> {
    if !input.exists() { return Err(PdfError::InputMissing(input.display().to_string())); }
    let doc = lopdf::Document::load(input)?;
    let pages = doc.get_pages();
    let already_has_acroform = doc.catalog().ok()
        .and_then(|c| c.get(b"AcroForm").ok()).is_some();
    let mut report = DetectionReport {
        candidates: Vec::new(),
        pages_scanned: pages.len() as u32,
        already_has_acroform,
        warnings: Vec::new(),
    };
    for (page_no, _) in &pages {
        match detect_page(&doc, *page_no) {
            Ok(mut cs) => report.candidates.append(&mut cs),
            Err(e) => report.warnings.push(format!("page {page_no}: {e}")),
        }
    }
    // Deduplicate names by appending _2, _3, ...
    dedupe_names(&mut report.candidates);
    Ok(report)
}

pub fn detect_page(doc: &lopdf::Document, page: u32) -> Result<Vec<FieldCandidate>, PdfError> {
    let prims = geom::walk_page(doc, page)?;
    let runs = text::walk_page(doc, page)?;
    let mut out = Vec::new();

    // (1) Empty boxes → checkbox candidates
    for r in &prims.rects {
        let w = r.x1 - r.x0; let h = r.y1 - r.y0;
        if w >= 8.0 && w <= 22.0 && h >= 8.0 && h <= 22.0 && (w - h).abs() < 2.0 {
            let label = nearest_label_to_the_right(&runs, r.x1, (r.y0 + r.y1) / 2.0, 100.0);
            out.push(FieldCandidate {
                suggested_name: slugify(label.as_deref().unwrap_or("checkbox")),
                kind: CandidateKind::Checkbox,
                page,
                rect: [r.x0, r.y0, r.x1, r.y1],
                label,
                evidence: Evidence::EmptyBox { side_pt: w },
                confidence: 0.7,
            });
        }
    }

    // (2) Horizontal rules → Text / Signature / LabeledBlank
    for line in &prims.lines {
        let dy = (line.y1 - line.y0).abs();
        let dx = (line.x1 - line.x0).abs();
        if dy < 1.0 && dx > 50.0 && line.width < 2.0 {
            let mid_y = (line.y0 + line.y1) / 2.0;
            let label = label_above_or_left(&runs, line.x0, line.x1, mid_y, 4.0, 40.0);
            let is_sig = label.as_deref()
                .map(|s| is_signature_label(s)).unwrap_or(false);
            let kind = if is_sig {
                CandidateKind::Signature
            } else {
                CandidateKind::Text { multiline: false }
            };
            let evidence = if is_sig {
                Evidence::SignatureLine
            } else if label.is_some() {
                Evidence::LabeledBlank
            } else {
                Evidence::HorizontalRule { line_width_pt: line.width }
            };
            let mut confidence = 0.55;
            if dx > 80.0 { confidence += 0.15; }
            if label.is_some() { confidence += 0.20; }
            let label_clean = label.as_deref().map(strip_trailing_colon).map(str::to_string);
            out.push(FieldCandidate {
                suggested_name: slugify(label_clean.as_deref().unwrap_or("text")),
                kind,
                page,
                rect: [line.x0, line.y0 - 2.0, line.x1, line.y0 + 14.0],
                label: label_clean,
                evidence,
                confidence,
            });
        }
    }
    Ok(out)
}

fn slugify(s: &str) -> String {
    let mut out = String::with_capacity(s.len());
    let mut prev_underscore = false;
    for ch in s.chars() {
        if ch.is_ascii_alphanumeric() { out.push(ch.to_ascii_lowercase()); prev_underscore = false; }
        else if !prev_underscore && !out.is_empty() { out.push('_'); prev_underscore = true; }
    }
    while out.ends_with('_') { out.pop(); }
    if out.is_empty() { out.push_str("field"); }
    out
}

fn strip_trailing_colon(s: &str) -> &str { s.trim_end_matches(':').trim() }

fn is_signature_label(s: &str) -> bool {
    let s = s.to_lowercase();
    s.contains("signature") || s.contains("sign here") || s.contains("signed by")
        || s.trim() == "x" || s.trim() == "x:"
}

fn nearest_label_to_the_right(runs: &[text::TextRun], x: f32, y: f32, max_dx: f32) -> Option<String> {
    runs.iter()
        .filter(|r| (r.y - y).abs() < 4.0 && r.x > x && (r.x - x) < max_dx)
        .min_by(|a, b| a.x.partial_cmp(&b.x).unwrap())
        .map(|r| strip_trailing_colon(&r.text).to_string())
}

fn label_above_or_left(runs: &[text::TextRun], lx0: f32, lx1: f32, ly: f32,
                       max_above: f32, max_right_gap: f32) -> Option<String> {
    // A label that ends just to the LEFT of the rule on (roughly) the same baseline.
    let same_line = runs.iter()
        .filter(|r| (r.y - ly).abs() <= max_above
                 && r.x < lx0
                 && (lx0 - r.x - approx_width(&r.text, r.size)) < max_right_gap)
        .max_by(|a, b| a.x.partial_cmp(&b.x).unwrap())
        .map(|r| r.text.clone());
    same_line
}

fn approx_width(s: &str, size: f32) -> f32 { (s.chars().count() as f32) * size * 0.5 }

fn dedupe_names(cands: &mut [FieldCandidate]) {
    use std::collections::HashMap;
    let mut counts: HashMap<String, u32> = HashMap::new();
    for c in cands.iter_mut() {
        let n = counts.entry(c.suggested_name.clone()).or_insert(0);
        *n += 1;
        if *n > 1 { c.suggested_name = format!("{}_{}", c.suggested_name, n); }
    }
}

```svelte
<script lang="ts">
  import { autodetect, type FieldCandidate } from '$lib/api/formsDetect';
  import { designAdd } from '$lib/api/formsDesign';
  import { fade } from 'svelte/transition';

  let { open = $bindable(false), inputPath } = $props<{ open: boolean; inputPath: string }>();
  let report = $state<null | Awaited<ReturnType<typeof autodetect>>>(null);
  let keep = $state(new Set<number>());
  let threshold = $state(0.7);
  let busy = $state(false);

  async function load() {
    busy = true;
    report = await autodetect(inputPath);
    keep = new Set(report.candidates.map((c, i) => c.confidence >= threshold ? i : -1).filter(i => i >= 0));
    busy = false;
  }
  $effect(() => { if (open && inputPath) load(); });

  async function commit() {
    if (!report) return;
    busy = true;
    const drafts = [...keep].map(i => candidateToDraft(report!.candidates[i]));
    await designAdd({ input: inputPath, output: inputPath, drafts });
    busy = false;
    open = false;
    // toast: `Added ${drafts.length} fields — switch to Fill view to test`
  }
</script>

{#if open}
  <div class="modal-backdrop" transition:fade>
    <div class="modal">
      <header><h2>Auto-detect form fields</h2></header>
      … (candidate table with confidence pills + checkboxes)
      <footer>
        <input type="range" min="0" max="1" step="0.05" bind:value={threshold} />
        <button onclick={commit} disabled={busy || keep.size === 0}>
          Add {keep.size} fields
        </button>
      </footer>
    </div>
  </div>
{/if}
```
