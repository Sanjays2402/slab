**Goal:** Close the final ISO 14289-1 gap in the v3.1.0 Loom pipeline by
emitting (a) a PDF/UA-1 XMP packet with the `pdfuaid:part=1` claim, (b) a
`/Metadata` catalog stream pointing at that packet, (c) `/ViewerPreferences
<< /DisplayDocTitle true >>`, and (d) a Document Info `/Title` mirroring the
XMP `dc:title`. Then layer a **post-tag PDF/UA validator UI** on top: a Loom
panel tab that runs the already-shipped Matterhorn auto-conditions against
the tagged output and renders a green/yellow/red verdict card with the failing
conditions itemised, linking each failure to the relevant Matterhorn section.

When this slice merges, `feature/v3.1.0-loom-slice-3` is feature-complete and
ready for `main` merge → tag `v3.1.0` → release.

**Architecture:** One new backend module
`src-tauri/src/pdf/loom/metadata.rs` (parallels `structure_tree.rs`),
plus a Tauri command `slab_loom_validate_pdfua` that drives
`matterhorn::auto_conditions()` against a freshly-loaded `lopdf::Document`
and returns a structured verdict. The Loom panel gets a new **Validate** tab
that, after a successful `Tag Document` run, calls the validator on the
emitted `.tagged.pdf` and shows the verdict inline. The flow becomes one
continuous in-app journey: open → tag → validate → ship.

**Tech Stack:**
- `lopdf` 0.36 (already a dep) for catalog + stream manipulation.
- `chrono` (already a dep) for ISO-8601 timestamps in XMP.
- Zero new crates. Hand-written XMP packet (mirrors `pdf/pdfa/xmp.rs`
  pattern — XMP is whitespace-sensitive and a generic XML serialiser will
  silently break conformance).

**Tech debt acknowledged:** `src-tauri/src/pdf/pdfa/xmp.rs` already emits a
`pdfaid` (PDF/A) packet. We deliberately ship a separate `loom::metadata`
module rather than overloading the PDF/A builder, because PDF/UA-1 and PDF/A
ship distinct XMP grammars (`pdfuaid:part` vs `pdfaid:part` + different
namespaces) and the eventual PDF/UA+A combo packet (v3.2.0 Press territory)
will pick the right composition then. **DRY-by-shared-helper, not
DRY-by-conditional**: both modules will eventually call a shared
`pdf/xmp/builder.rs` that emits the RDF/XML scaffolding — that refactor is a
follow-up issue, not part of this slice.

```rust
//! PDF/UA-1 metadata stamping for Slab Loom.
//!
//! ISO 14289-1 §7 requires four catalog-level annotations beyond the
//! StructTreeRoot already emitted by `structure_tree::weave`:
//!
//!   1. An XMP metadata stream attached via `/Metadata` carrying the
//!      `pdfuaid:part=1` claim (the only formal conformance marker).
//!   2. `/ViewerPreferences << /DisplayDocTitle true >>` so assistive
//!      tech reads `dc:title` instead of the filename.
//!   3. A Document Info `/Title` mirroring the XMP `dc:title` (legacy
//!      readers fall back to this).
//!   4. `/Lang` already set by `structure_tree::weave`.
//!
//! `stamp(doc, opts)` applies all four; `build_pdfua_xmp(opts)` is the
//! pure byte-level helper used internally and exposed for tests.

use chrono::{DateTime, Utc};
use lopdf::{Dictionary, Document, Object, Stream};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct MetadataOptions {
    pub title: Option<String>,
    pub author: Option<String>,
    pub language: Option<String>,
    pub create_date: Option<DateTime<Utc>>,
    pub modify_date: Option<DateTime<Utc>>,
    pub producer: Option<String>,
}

#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct MetadataStats {
    pub xmp_bytes: usize,
    pub display_doc_title_set: bool,
    pub info_title_set: bool,
}

const DEFAULT_PRODUCER: &str = "Slab Loom v3.1.0";

/// Pure helper: assemble the XMP packet bytes for a PDF/UA-1 document.
pub fn build_pdfua_xmp(opts: &MetadataOptions) -> Vec<u8> {
    let producer = opts.producer.as_deref().unwrap_or(DEFAULT_PRODUCER);
    let create = opts.create_date.unwrap_or_else(Utc::now);
    let modify = opts.modify_date.unwrap_or(create);
    let create_iso = create.format("%Y-%m-%dT%H:%M:%SZ").to_string();
    let modify_iso = modify.format("%Y-%m-%dT%H:%M:%SZ").to_string();

    let mut s = String::with_capacity(2048);
    s.push_str("<?xpacket begin=\"\u{FEFF}\" id=\"W5M0MpCehiHzreSzNTczkc9d\"?>\n");
    s.push_str("<x:xmpmeta xmlns:x=\"adobe:ns:meta/\" x:xmptk=\"Slab Loom 3.1.0\">\n");
    s.push_str("  <rdf:RDF xmlns:rdf=\"http://www.w3.org/1999/02/22-rdf-syntax-ns#\">\n");

    // pdfuaid (the mandatory PDF/UA-1 claim).
    s.push_str(
        "    <rdf:Description rdf:about=\"\" xmlns:pdfuaid=\"http://www.aiim.org/pdfua/ns/id/\">\n",
    );
    s.push_str("      <pdfuaid:part>1</pdfuaid:part>\n");
    s.push_str("    </rdf:Description>\n");

    // Dublin Core (title + creator) — skipped entirely if neither is set.
    if opts.title.is_some() || opts.author.is_some() {
        s.push_str(
            "    <rdf:Description rdf:about=\"\" xmlns:dc=\"http://purl.org/dc/elements/1.1/\">\n",
        );
        if let Some(t) = &opts.title {
            s.push_str("      <dc:title><rdf:Alt><rdf:li xml:lang=\"x-default\">");
            s.push_str(&xml_escape(t));
            s.push_str("</rdf:li></rdf:Alt></dc:title>\n");
        }
        if let Some(a) = &opts.author {
            s.push_str("      <dc:creator><rdf:Seq><rdf:li>");
            s.push_str(&xml_escape(a));
            s.push_str("</rdf:li></rdf:Seq></dc:creator>\n");
        }
        s.push_str("    </rdf:Description>\n");
    }

    // XMP basic.
    s.push_str("    <rdf:Description rdf:about=\"\" xmlns:xmp=\"http://ns.adobe.com/xap/1.0/\">\n");
    s.push_str(&format!("      <xmp:CreateDate>{create_iso}</xmp:CreateDate>\n"));
    s.push_str(&format!("      <xmp:ModifyDate>{modify_iso}</xmp:ModifyDate>\n"));
    s.push_str(&format!(
        "      <xmp:CreatorTool>{}</xmp:CreatorTool>\n",
        xml_escape(producer)
    ));
    s.push_str("    </rdf:Description>\n");

    // PDF producer (legacy compat).
    s.push_str("    <rdf:Description rdf:about=\"\" xmlns:pdf=\"http://ns.adobe.com/pdf/1.3/\">\n");
    s.push_str(&format!(
        "      <pdf:Producer>{}</pdf:Producer>\n",
        xml_escape(producer)
    ));
    s.push_str("    </rdf:Description>\n");

    s.push_str("  </rdf:RDF>\n");
    s.push_str("</x:xmpmeta>\n");

    // ~2KB whitespace padding for in-place growth (Adobe XMP §7.3.2).
    for _ in 0..40 {
        s.push_str("                                                  \n");
    }
    s.push_str("<?xpacket end=\"w\"?>");
    s.into_bytes()
}

fn xml_escape(input: &str) -> String {
    let mut out = String::with_capacity(input.len());
    for ch in input.chars() {
        match ch {
            '<' => out.push_str("&lt;"),
            '>' => out.push_str("&gt;"),
            '&' => out.push_str("&amp;"),
            '\'' => out.push_str("&apos;"),
            '"' => out.push_str("&quot;"),
            c if (c as u32) < 0x20 && c != '\n' && c != '\t' && c != '\r' => out.push(' '),
            c => out.push(c),
        }
    }
    out
}
```

```rust
/// Apply PDF/UA-1 metadata stamping in place. Returns stats.
pub fn stamp(doc: &mut Document, opts: &MetadataOptions) -> Result<MetadataStats, String> {
    let mut stats = MetadataStats::default();

    // 1. Build XMP packet.
    let xmp_bytes = build_pdfua_xmp(opts);
    stats.xmp_bytes = xmp_bytes.len();

    // 2. Allocate /Metadata stream — uncompressed XML per ISO 14289-1 §7.1.
    let mut meta_dict = Dictionary::new();
    meta_dict.set("Type", Object::Name(b"Metadata".to_vec()));
    meta_dict.set("Subtype", Object::Name(b"XML".to_vec()));
    let meta_stream = Stream::new(meta_dict, xmp_bytes).with_compression(false);
    let meta_id = doc.add_object(Object::Stream(meta_stream));

    // 3. Catalog wiring.
    let cat_id = doc.catalog().map_err(|e| format!("catalog: {e}"))?;
    let cat = doc
        .get_object_mut(cat_id)
        .and_then(|o| o.as_dict_mut())
        .map_err(|e| format!("catalog dict: {e}"))?;
    cat.set("Metadata", Object::Reference(meta_id));

    // ViewerPreferences << /DisplayDocTitle true >> — preserve existing entries.
    let mut vp = match cat.get(b"ViewerPreferences").ok().and_then(|o| o.as_dict().ok()) {
        Some(existing) => existing.clone(),
        None => Dictionary::new(),
    };
    vp.set("DisplayDocTitle", Object::Boolean(true));
    cat.set("ViewerPreferences", Object::Dictionary(vp));
    stats.display_doc_title_set = true;

    // 4. Document Info /Title (legacy fallback).
    if let Some(t) = &opts.title {
        let info_id = match doc.trailer.get(b"Info").ok().and_then(|o| o.as_reference().ok()) {
            Some(id) => id,
            None => {
                let id = doc.add_object(Dictionary::new());
                doc.trailer.set("Info", Object::Reference(id));
                id
            }
        };
        if let Ok(Object::Dictionary(info)) = doc.get_object_mut(info_id) {
            info.set("Title", Object::string_literal(t.clone()));
            stats.info_title_set = true;
        }
    }

    Ok(stats)
}
```

```rust
//! PDF/UA-1 post-tag validator — runs Matterhorn 1.1 auto conditions
//! against a tagged lopdf::Document.

use crate::pdf::loom::matterhorn::{auto_conditions, FailureCondition};
use lopdf::Document;
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ValidationFailure {
    pub condition_id: String,
    pub section_id: String,
    pub title: String,
    pub detail: String,
}

#[derive(Debug, Clone, Default, Serialize, Deserialize)]
pub struct ValidationReport {
    pub passed: usize,
    pub skipped: usize,
    pub failed: Vec<ValidationFailure>,
    pub total_auto: usize,
    pub total_human: usize,
}

pub fn validate(doc: &Document) -> ValidationReport {
    let mut rep = ValidationReport::default();
    let auto: Vec<&FailureCondition> = auto_conditions().collect();
    rep.total_auto = auto.len();
    rep.total_human = crate::pdf::loom::matterhorn::human_conditions().count();

    for cond in auto {
        match dispatch(cond, doc) {
            CheckOutcome::Pass => rep.passed += 1,
            CheckOutcome::Skip => rep.skipped += 1,
            CheckOutcome::Fail(detail) => rep.failed.push(ValidationFailure {
                condition_id: cond.id.to_string(),
                section_id: cond.section_id.to_string(),
                title: cond.title.to_string(),
                detail,
            }),
        }
    }
    rep
}

enum CheckOutcome {
    Pass,
    Skip,
    Fail(String),
}

fn dispatch(cond: &FailureCondition, doc: &Document) -> CheckOutcome {
    match cond.id {
        // Look up real IDs via the registry — examples shown.
        "07-001" => check_struct_tree_root(doc),
        "07-002" => check_mark_info_marked(doc),
        "07-003" => check_lang(doc),
        "11-001" => check_metadata_stream(doc),
        "11-002" => check_pdfuaid_part(doc),
        "11-006" => check_display_doc_title(doc),
        _ => CheckOutcome::Skip, // not yet implemented in v3.1.0
    }
}

fn check_struct_tree_root(doc: &Document) -> CheckOutcome {
    let Ok(cat_id) = doc.catalog() else { return CheckOutcome::Fail("no catalog".into()) };
    let cat = match doc.get_object(cat_id).and_then(|o| o.as_dict()) {
        Ok(d) => d,
        Err(_) => return CheckOutcome::Fail("catalog not a dict".into()),
    };
    if cat.has(b"StructTreeRoot") { CheckOutcome::Pass } else {
        CheckOutcome::Fail("/StructTreeRoot missing on catalog".into())
    }
}
```

```rust
pub struct LoomValidateResult {
    pub passed: usize,
    pub skipped: usize,
    pub failed: Vec<crate::pdf::loom::validate::ValidationFailure>,
    pub total_auto: usize,
    pub total_human: usize,
    pub elapsed_ms: u64,
}

#[tauri::command]
async fn slab_loom_validate_pdfua(input: PathBuf) -> CmdResult<LoomValidateResult> {
    use crate::pdf::loom::validate::validate;
    use lopdf::Document;
    let started = std::time::Instant::now();
    let doc = match Document::load(&input) {
        Ok(d) => d,
        Err(e) => return CmdResult::Err { message: format!("load {}: {}", input.display(), e) },
    };
    let rep = validate(&doc);
    CmdResult::Ok {
        data: LoomValidateResult {
            passed: rep.passed,
            skipped: rep.skipped,
            failed: rep.failed,
            total_auto: rep.total_auto,
            total_human: rep.total_human,
            elapsed_ms: started.elapsed().as_millis() as u64,
        },
    }
}
```
