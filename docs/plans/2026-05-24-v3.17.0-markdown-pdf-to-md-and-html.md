**Goal:** Add two new export targets to Slab — PDF → Markdown (`.md`) and
PDF → semantic HTML (`.html`) — implemented by writing two new emitters that
share Slab's existing `Block` IR with the v3.14.0 Reflow pipeline. Ship both
as standalone panels and as Atelier batch steps, end-to-end, in one release.

**Architecture:** The Reflow pipeline (`src-tauri/src/pdf/reflow/`) already
turns a PDF into `Vec<Block>` (Body / Heading / ListItem / TableRow). Add a
new top-level module `src-tauri/src/pdf/markdown/` that re-uses
`reflow::extract` + `reflow::layout` and emits Markdown via a `md.rs` writer
and HTML via an `html.rs` writer. Both writers consume `&[Block]` and produce
a `String` (no zip, no XML schema work). Then surface two Tauri commands,
two new Svelte panels, two nav entries, two Atelier steps, and a unified
preview pane.

**Tech Stack:** Rust (`lopdf` 0.40 already in deps, `pulldown-cmark` for
optional round-trip validation in tests, hand-rolled emitters — no new heavy
deps). Frontend Svelte 5 + Tauri 2. Atelier integration via the existing
`Step` enum machinery.

use super::types::{MarkdownFlavour, MarkdownOptions};
use crate::pdf::reflow::types::{Block, ListKind};
use std::fmt::Write;

/// Convert a Block IR slice into a Markdown string.
pub fn emit_markdown(blocks: &[Block], opts: &MarkdownOptions) -> String {
    let mut out = String::with_capacity(blocks.len() * 64);
    let mut table_buf: Vec<Vec<String>> = Vec::new();

    for block in blocks {
        // Flush pending table on any non-TableRow block.
        if !matches!(block, Block::TableRow { .. }) && !table_buf.is_empty() {
            flush_table(&mut out, &table_buf, opts);
            table_buf.clear();
        }

        match block {
            Block::Body { text } => {
                writeln!(out, "{}\n", escape(text)).ok();
            }
            Block::Heading { level, text } => {
                let hashes = "#".repeat((*level as usize).clamp(1, 6));
                writeln!(out, "{} {}\n", hashes, escape(text)).ok();
            }
            Block::ListItem { kind, level, text } => {
                let indent = "  ".repeat(*level as usize);
                let marker = match kind {
                    ListKind::Bullet => "-",
                    ListKind::Number => "1.",
                };
                writeln!(out, "{}{} {}", indent, marker, escape(text)).ok();
            }
            Block::TableRow { cells } => {
                table_buf.push(cells.clone());
            }
        }
    }

    if !table_buf.is_empty() {
        flush_table(&mut out, &table_buf, opts);
    }

    out
}

fn flush_table(out: &mut String, rows: &[Vec<String>], opts: &MarkdownOptions) {
    if rows.is_empty() {
        return;
    }
    match opts.flavour {
        MarkdownFlavour::Gfm => {
            let header = &rows[0];
            writeln!(
                out,
                "| {} |",
                header.iter().map(|c| escape_cell(c)).collect::<Vec<_>>().join(" | ")
            ).ok();
            writeln!(
                out,
                "| {} |",
                vec!["---"; header.len()].join(" | ")
            ).ok();
            for row in &rows[1..] {
                writeln!(
                    out,
                    "| {} |",
                    row.iter().map(|c| escape_cell(c)).collect::<Vec<_>>().join(" | ")
                ).ok();
            }
            out.push('\n');
        }
        MarkdownFlavour::CommonMark => {
            out.push_str("```\n");
            for row in rows {
                writeln!(out, "{}", row.join("\t")).ok();
            }
            out.push_str("```\n\n");
        }
    }
}

fn escape(text: &str) -> String {
    let mut s = String::with_capacity(text.len());
    for c in text.chars() {
        match c {
            '\\' | '`' | '*' | '_' | '{' | '}' | '[' | ']' | '(' | ')' | '#' | '+' | '-' | '.' | '!' | '|' | '<' | '>' => {
                s.push('\\');
                s.push(c);
            }
            _ => s.push(c),
        }
    }
    s
}

fn escape_cell(text: &str) -> String {
    text.replace('|', r"\|").replace('\n', " ")
}
```

```rust
use super::types::HtmlOptions;
use crate::pdf::reflow::types::{Block, ListKind};
use std::fmt::Write;

const DEFAULT_CSS: &str = r#"
body{font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',sans-serif;max-width:780px;margin:2rem auto;padding:0 1rem;color:#111;line-height:1.6}
h1,h2,h3{color:#0a0a0a;margin-top:1.8em}
table{border-collapse:collapse;width:100%;margin:1em 0}
th,td{border:1px solid #ddd;padding:6px 10px;text-align:left}
th{background:#f6f6f6}
ul,ol{padding-left:1.4em}
article{padding:1rem 0}
"#;

pub fn emit_html(blocks: &[Block], opts: &HtmlOptions) -> String {
    let mut out = String::with_capacity(blocks.len() * 96);
    out.push_str("<!DOCTYPE html>\n<html lang=\"en\">\n<head>\n<meta charset=\"UTF-8\">\n");
    out.push_str("<meta name=\"generator\" content=\"Slab v3.17.0 Markdown\">\n");
    out.push_str("<title>Document</title>\n");
    if opts.embed_css {
        out.push_str("<style>");
        out.push_str(DEFAULT_CSS);
        out.push_str("</style>\n");
    }
    out.push_str("</head>\n<body>\n");
    if opts.semantic_tags {
        out.push_str("<article>\n");
    }

    let mut i = 0;
    let mut in_list: Option<ListKind> = None;
    while i < blocks.len() {
        // Close any open list when the next block isn't a ListItem.
        if !matches!(blocks[i], Block::ListItem { .. }) && in_list.is_some() {
            out.push_str(close_list_tag(in_list.unwrap()));
            in_list = None;
        }
        match &blocks[i] {
            Block::Body { text } => {
                writeln!(out, "<p>{}</p>", escape(text)).ok();
                i += 1;
            }
            Block::Heading { level, text } => {
                let lvl = (*level as usize).clamp(1, 6);
                writeln!(out, "<h{0}>{1}</h{0}>", lvl, escape(text)).ok();
                i += 1;
            }
            Block::ListItem { kind, text, .. } => {
                if in_list != Some(*kind) {
                    if let Some(prev) = in_list {
                        out.push_str(close_list_tag(prev));
                    }
                    out.push_str(open_list_tag(*kind));
                    in_list = Some(*kind);
                }
                writeln!(out, "  <li>{}</li>", escape(text)).ok();
                i += 1;
            }
            Block::TableRow { .. } => {
                // Collect all consecutive TableRows.
                let start = i;
                while i < blocks.len() && matches!(blocks[i], Block::TableRow { .. }) {
                    i += 1;
                }
                emit_table(&mut out, &blocks[start..i]);
            }
        }
    }
    if let Some(prev) = in_list {
        out.push_str(close_list_tag(prev));
    }

    if opts.semantic_tags {
        out.push_str("</article>\n");
    }
    out.push_str("</body>\n</html>\n");
    out
}

fn open_list_tag(k: ListKind) -> &'static str {
    match k {
        ListKind::Bullet => "<ul>\n",
        ListKind::Number => "<ol>\n",
    }
}

fn close_list_tag(k: ListKind) -> &'static str {
    match k {
        ListKind::Bullet => "</ul>\n",
        ListKind::Number => "</ol>\n",
    }
}

fn emit_table(out: &mut String, rows: &[Block]) {
    if rows.is_empty() {
        return;
    }
    let cells = |b: &Block| -> Vec<String> {
        if let Block::TableRow { cells } = b {
            cells.clone()
        } else {
            Vec::new()
        }
    };
    out.push_str("<table>\n<thead>\n<tr>");
    for c in cells(&rows[0]) {
        write!(out, "<th>{}</th>", escape(&c)).ok();
    }
    out.push_str("</tr>\n</thead>\n<tbody>\n");
    for row in &rows[1..] {
        out.push_str("<tr>");
        for c in cells(row) {
            write!(out, "<td>{}</td>", escape(&c)).ok();
        }
        out.push_str("</tr>\n");
    }
    out.push_str("</tbody>\n</table>\n");
}

fn escape(s: &str) -> String {
    let mut out = String::with_capacity(s.len());
    for c in s.chars() {
        match c {
            '<' => out.push_str("&lt;"),
            '>' => out.push_str("&gt;"),
            '&' => out.push_str("&amp;"),
            '"' => out.push_str("&quot;"),
            _ => out.push(c),
        }
    }
    out
}

```rust
pub fn convert_to_markdown(
    input: &Path,
    output: &Path,
    opts: &MarkdownOptions,
) -> Result<MarkdownReport, MarkdownError> {
    if !input.exists() {
        return Err(MarkdownError::InputMissing(input.display().to_string()));
    }
    if let Some(parent) = output.parent() {
        if !parent.as_os_str().is_empty() && !parent.exists() {
            return Err(MarkdownError::OutputNotWritable(output.display().to_string()));
        }
    }

    let started = Instant::now();
    let doc = lopdf::Document::load(input)?;
    let runs = crate::pdf::reflow::extract::extract_text_runs(&doc)?;
    let reflow_opts = crate::pdf::reflow::types::ReflowOptions {
        detect_tables: opts.detect_tables,
        detect_lists: opts.detect_lists,
        heading_size_ratio: opts.heading_size_ratio,
        preserve_page_breaks: opts.preserve_page_breaks,
        locale: "en".to_string(),
    };
    let blocks = crate::pdf::reflow::layout::reconstruct_blocks(&runs, &reflow_opts);

    let md = md::emit_markdown(&blocks, opts);
    let bytes = md.into_bytes();
    std::fs::write(output, &bytes)?;

    Ok(MarkdownReport {
        headings: count(&blocks, |b| matches!(b, crate::pdf::reflow::types::Block::Heading { .. })),
        paragraphs: count(&blocks, |b| matches!(b, crate::pdf::reflow::types::Block::Body { .. })),
        list_items: count(&blocks, |b| matches!(b, crate::pdf::reflow::types::Block::ListItem { .. })),
        tables: count_tables(&blocks),
        bytes_written: bytes.len() as u64,
        duration_ms: started.elapsed().as_millis() as u64,
    })
}

pub fn convert_to_html(
    input: &Path,
    output: &Path,
    opts: &HtmlOptions,
) -> Result<HtmlReport, MarkdownError> {
    if !input.exists() {
        return Err(MarkdownError::InputMissing(input.display().to_string()));
    }
    if let Some(parent) = output.parent() {
        if !parent.as_os_str().is_empty() && !parent.exists() {
            return Err(MarkdownError::OutputNotWritable(output.display().to_string()));
        }
    }

    let started = Instant::now();
    let doc = lopdf::Document::load(input)?;
    let runs = crate::pdf::reflow::extract::extract_text_runs(&doc)?;
    let reflow_opts = crate::pdf::reflow::types::ReflowOptions {
        detect_tables: opts.detect_tables,
        detect_lists: opts.detect_lists,
        heading_size_ratio: opts.heading_size_ratio,
        preserve_page_breaks: false,
        locale: "en".to_string(),
    };
    let blocks = crate::pdf::reflow::layout::reconstruct_blocks(&runs, &reflow_opts);

    let html = html::emit_html(&blocks, opts);
    let bytes = html.into_bytes();
    std::fs::write(output, &bytes)?;

    Ok(HtmlReport {
        headings: count(&blocks, |b| matches!(b, crate::pdf::reflow::types::Block::Heading { .. })),
        paragraphs: count(&blocks, |b| matches!(b, crate::pdf::reflow::types::Block::Body { .. })),
        list_items: count(&blocks, |b| matches!(b, crate::pdf::reflow::types::Block::ListItem { .. })),
        tables: count_tables(&blocks),
        bytes_written: bytes.len() as u64,
        duration_ms: started.elapsed().as_millis() as u64,
    })
}

fn count(blocks: &[crate::pdf::reflow::types::Block], pred: impl Fn(&crate::pdf::reflow::types::Block) -> bool) -> u32 {
    blocks.iter().filter(|b| pred(b)).count() as u32
}

fn count_tables(blocks: &[crate::pdf::reflow::types::Block]) -> u32 {
    let mut tables = 0u32;
    let mut in_table = false;
    for b in blocks {
        let is_row = matches!(b, crate::pdf::reflow::types::Block::TableRow { .. });
        if is_row && !in_table {
            tables += 1;
            in_table = true;
        } else if !is_row {
            in_table = false;
        }
    }
    tables
}
