# v3.37.0 "Atlas Smart Folders Hub" Implementation Plan

> **For Hermes (cron):** Use `subagent-driven-development` skill to ship this plan task-by-task. Off-hours only. Honor SHIP-SIZE rules: ≥4 commits, ≥600 net LOC, end-to-end working.

**Goal:** Ship a unified "Smart Folders" hub — one dedicated panel that lists every built-in preset and every personal `.slabpresets`-imported preset side-by-side, with drag-to-reorder, one-click apply, search, and bulk export.

**Architecture:** New Tauri command `slab_smart_folders_list` merges built-in `PresetInfo` (from `presets.rs`) and `PersonalPresetRecord` (from `personal_presets.rs`) into a single ordered `SmartFolderEntry` enum, persisted via a new `library_smart_folder_order` table. New Svelte panel `SmartFoldersHubPanel.svelte` renders the merged list with HTML5 drag-and-drop, registers in the nav rail, Cmd+K palette, keyboard shortcut hub (Cmd+Shift+F), and settings.

**Tech Stack:** Rust (rusqlite + serde), Tauri 2 commands, Svelte 5 (runes), existing `library.ts` API surface.

**Wedge:** Acrobat has nothing like this. PDF Expert has rigid built-in tags. Slab gives a 200-lawyer firm one screen to manage 6 built-ins + 40 custom org-wide presets, drag them into the order paralegals want, and export the whole pack as a single `.slabpresets` file IT distributes via MDM.

```rust
//! v3.37.0 — Atlas Smart Folders Hub.
//! Merges built-in `Preset` definitions with personal presets into one
//! ordered, pinnable list backed by `library_smart_folder_order`.

use serde::{Deserialize, Serialize};

use super::personal_presets::list_personal_presets;
use super::presets::list_presets;
use super::registry::{LibraryDb, LibraryError};

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub struct SmartFolderEntry {
    pub kind: String,            // "builtin" | "personal"
    pub id: String,              // preset id (builtin str OR personal i64 as text)
    pub name: String,
    pub icon: String,
    pub color: String,
    pub description: String,
    pub pinned: bool,
    pub sort_order: i64,
}

pub fn list_smart_folders(db: &mut LibraryDb) -> Result<Vec<SmartFolderEntry>, LibraryError> {
    // 1. Gather sources.
    let builtins = list_presets();
    let personal = list_personal_presets(db)?;

    // 2. Load any persisted (order, pinned) tuples.
    let conn = db.conn();
    let mut stmt = conn.prepare(
        "SELECT entry_kind, entry_id, sort_order, pinned
           FROM library_smart_folder_order",
    )?;
    let rows: Vec<(String, String, i64, bool)> = stmt
        .query_map([], |r| {
            Ok((
                r.get::<_, String>(0)?,
                r.get::<_, String>(1)?,
                r.get::<_, i64>(2)?,
                r.get::<_, i64>(3)? != 0,
            ))
        })?
        .collect::<Result<_, _>>()?;
    drop(stmt);

    let lookup = |kind: &str, id: &str| -> (i64, bool) {
        rows.iter()
            .find(|(k, i, _, _)| k == kind && i == id)
            .map(|(_, _, s, p)| (*s, *p))
            .unwrap_or((i64::MAX, false))
    };

    let mut out: Vec<SmartFolderEntry> = Vec::new();
    for (idx, b) in builtins.iter().enumerate() {
        let (order, pinned) = lookup("builtin", &b.id);
        out.push(SmartFolderEntry {
            kind: "builtin".into(),
            id: b.id.clone(),
            name: b.name.clone(),
            icon: b.icon.clone(),
            color: b.color.clone(),
            description: b.description.clone(),
            pinned,
            sort_order: if order == i64::MAX { 1_000 + idx as i64 } else { order },
        });
    }
    for p in &personal {
        let id_str = p.id.to_string();
        let (order, pinned) = lookup("personal", &id_str);
        out.push(SmartFolderEntry {
            kind: "personal".into(),
            id: id_str,
            name: p.name.clone(),
            icon: p.icon.clone().unwrap_or_else(|| "star".into()),
            color: p.color.clone().unwrap_or_else(|| "#888".into()),
            description: p.description.clone().unwrap_or_default(),
            pinned,
            sort_order: if order == i64::MAX { 2_000 + p.sort_order } else { order },
        });
    }

    // 3. Sort: pinned-first, then sort_order, then name.
    out.sort_by(|a, b| {
        b.pinned
            .cmp(&a.pinned)
            .then(a.sort_order.cmp(&b.sort_order))
            .then(a.name.cmp(&b.name))
    });

    Ok(out)
}

pub fn set_order(
    db: &mut LibraryDb,
    items: &[(&str, &str, i64)],
) -> Result<(), LibraryError> {
    let tx = db.conn_mut().transaction()?;
    for (kind, id, order) in items {
        tx.execute(
            "INSERT INTO library_smart_folder_order (entry_kind, entry_id, sort_order, pinned)
             VALUES (?1, ?2, ?3, COALESCE((SELECT pinned FROM library_smart_folder_order
                                          WHERE entry_kind=?1 AND entry_id=?2), 0))
             ON CONFLICT(entry_kind, entry_id)
             DO UPDATE SET sort_order = excluded.sort_order",
            rusqlite::params![kind, id, order],
        )?;
    }
    tx.commit()?;
    Ok(())
}

pub fn set_pinned(
    db: &mut LibraryDb,
    kind: &str,
    id: &str,
    pinned: bool,
) -> Result<(), LibraryError> {
    let conn = db.conn();
    conn.execute(
        "INSERT INTO library_smart_folder_order (entry_kind, entry_id, sort_order, pinned)
         VALUES (?1, ?2, COALESCE((SELECT sort_order FROM library_smart_folder_order
                                   WHERE entry_kind=?1 AND entry_id=?2), 0), ?3)
         ON CONFLICT(entry_kind, entry_id)
         DO UPDATE SET pinned = excluded.pinned",
        rusqlite::params![kind, id, pinned as i64],
    )?;
    Ok(())
}
```

```svelte
<script lang="ts">
  import { onMount } from "svelte";
  import {
    smartFoldersList, smartFoldersReorder, smartFoldersPin,
    presetApply, personalPresetApply, personalPresetsExport,
    type SmartFolderEntry,
  } from "$lib/library";

  let entries = $state<SmartFolderEntry[]>([]);
  let query = $state("");
  let loading = $state(true);
  let dragId = $state<string | null>(null);

  const visible = $derived(
    entries.filter(e =>
      query === "" ||
      e.name.toLowerCase().includes(query.toLowerCase()) ||
      e.description.toLowerCase().includes(query.toLowerCase()),
    ),
  );

  async function refresh() {
    loading = true;
    try { entries = await smartFoldersList(); }
    finally { loading = false; }
  }

  onMount(refresh);

  function key(e: SmartFolderEntry) { return `${e.kind}:${e.id}`; }

  async function onDrop(target: SmartFolderEntry) {
    if (!dragId) return;
    const from = entries.findIndex(e => key(e) === dragId);
    const to = entries.findIndex(e => key(e) === key(target));
    if (from < 0 || to < 0 || from === to) return;
    const moved = entries.slice();
    const [item] = moved.splice(from, 1);
    moved.splice(to, 0, item);
    entries = moved;
    dragId = null;
    // Persist new order.
    await smartFoldersReorder(
      moved.map((e, i) => [e.kind, e.id, i] as [string, string, number]),
    );
  }

  async function togglePin(e: SmartFolderEntry) {
    await smartFoldersPin(e.kind, e.id, !e.pinned);
    await refresh();
  }

  async function apply(e: SmartFolderEntry) {
    if (e.kind === "builtin") await presetApply(e.id);
    else await personalPresetApply(Number(e.id));
  }

  async function exportAllPersonal() {
    const personalIds = entries.filter(e => e.kind === "personal").map(e => Number(e.id));
```
