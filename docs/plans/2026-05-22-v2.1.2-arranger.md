**Architecture:**

```rust
use std::io::Write;
use std::path::Path;

/// Atomic write: stage to `<target>.slab-tmp.<pid>`, fsync, rename(2) over
/// the target. POSIX rename(2) is atomic on the same filesystem, so a crash
/// between `write` and `rename` leaves the original file untouched.
pub fn atomic_save(target: &Path, bytes: &[u8]) -> Result<(), PdfError> {
    let parent = target.parent().ok_or_else(|| {
        PdfError::Other("atomic_save target has no parent dir".into())
    })?;
    let tmp = parent.join(format!(
        ".slab-tmp.{}.{}",
        std::process::id(),
        target.file_name().and_then(|s| s.to_str()).unwrap_or("out")
    ));
    {
        let mut f = std::fs::File::create(&tmp)?;
        f.write_all(bytes)?;
        f.sync_all()?;
    }
    std::fs::rename(&tmp, target)?;
    Ok(())
}
```

```rust
/// Bake rotation into page geometry. Strips the `/Rotate` entry, swaps the
/// `/MediaBox` and `/CropBox` (for 90°/270°), and wraps every content stream
/// in a `q ... Q` matrix that performs the actual coordinate rotation. The
/// result re-opens identically rotated in any PDF viewer.
pub fn rotate_pages_permanent(
    input: &Path,
    pages: &[u32],
    rot: Rotation,
    output: &Path,
) -> Result<u32, PdfError> {
    let mut doc = Document::load(input)?;
    let page_map = doc.get_pages();
    let total = page_map.len() as u32;
    let targets: BTreeSet<u32> = if pages.is_empty() {
        (1..=total).collect()
    } else {
        for &p in pages {
            if p == 0 || p > total {
                return Err(PdfError::Other(format!(
                    "page {} out of range (1..={})", p, total
                )));
            }
        }
        pages.iter().copied().collect()
    };

    let mut count = 0u32;
    let page_ids: Vec<(u32, lopdf::ObjectId)> = page_map.into_iter().collect();
    for (page_num, page_id) in page_ids {
        if !targets.contains(&page_num) { continue; }
        bake_rotation_on_page(&mut doc, page_id, rot)?;
        count += 1;
    }

    // Reuse atomic_save for crash safety.
    let mut buf: Vec<u8> = Vec::new();
    doc.save_to(&mut buf)?;
    crate::pdf::atomic_save(output, &buf)?;
    Ok(count)
}

fn bake_rotation_on_page(
    doc: &mut Document,
    page_id: lopdf::ObjectId,
    rot: Rotation,
) -> Result<(), PdfError> {
    // 1. Read MediaBox (inherit-aware), strip existing /Rotate.
    let (w, h) = read_mediabox_inheriting(doc, page_id)?;

    // 2. Compute new MediaBox dims + the rotation matrix.
    //    PDF coordinates: origin at bottom-left, units = points.
    let (nw, nh, matrix) = match rot {
        Rotation::Cw90 => (h, w, format!("0 -1 1 0 0 {} cm", w)),
        Rotation::Cw180 => (w, h, format!("-1 0 0 -1 {} {} cm", w, h)),
        Rotation::Cw270 => (h, w, format!("0 1 -1 0 {} 0 cm", h)),
    };

    // 3. Prepend `q <matrix>` and append ` Q` to all content streams.
    let page = doc.get_object_mut(page_id)?.as_dict_mut()?;
    let contents_obj = page.get(b"Contents").ok();
    let new_box = vec![0.into(), 0.into(),
        Object::Real(nw as f32), Object::Real(nh as f32)];
    page.set("MediaBox", new_box.clone());
    if page.has(b"CropBox") {
        page.set("CropBox", new_box);
    }
    page.remove(b"Rotate");

    if let Some(c) = contents_obj.cloned() {
        let stream_ids: Vec<lopdf::ObjectId> = match c {
            Object::Reference(id) => vec![id],
            Object::Array(arr) => arr.iter().filter_map(|o| o.as_reference().ok()).collect(),
            _ => Vec::new(),
        };
        for sid in stream_ids {
            let s = doc.get_object_mut(sid)?.as_stream_mut()?;
            let mut new_content = format!("q {} ", matrix).into_bytes();
            new_content.extend_from_slice(&s.content);
            new_content.extend_from_slice(b" Q");
            s.set_content(new_content);
        }
    }
    Ok(())
}

fn read_mediabox_inheriting(
    doc: &Document,
    page_id: lopdf::ObjectId,
) -> Result<(f32, f32), PdfError> {
    let mut node_id = page_id;
    loop {
        let dict = doc.get_object(node_id)?.as_dict()?;
        if let Ok(arr) = dict.get(b"MediaBox").and_then(|o| o.as_array()) {
            let llx = arr[0].as_f32().unwrap_or(0.0);
            let lly = arr[1].as_f32().unwrap_or(0.0);
            let urx = arr[2].as_f32().unwrap_or(0.0);
            let ury = arr[3].as_f32().unwrap_or(0.0);
            return Ok((urx - llx, ury - lly));
        }
        node_id = dict.get(b"Parent").and_then(|o| o.as_reference())?;
    }
}
```

```rust
) -> Result<Vec<lopdf::ObjectId>, PdfError> {
    let img = image::open(img_path)
        .map_err(|e| PdfError::Other(format!("image decode: {e}")))?;
    let rgb = img.to_rgb8();
    let (iw, ih) = (rgb.width(), rgb.height());

    // Convert pixels → PDF points (72 pt = 1 in). dpi == pixels per inch.
    let dpi = if dpi <= 0.0 { 72.0 } else { dpi };
    let pw = (iw as f32) * 72.0 / dpi;
    let ph = (ih as f32) * 72.0 / dpi;

    // Build an /XObject /Image with FlateDecode'd raw DeviceRGB bytes.
    let raw = rgb.into_raw();
    let compressed = flate_compress(&raw);
    let mut img_stream = lopdf::Stream::new(
        dictionary! {
            "Type" => "XObject",
            "Subtype" => "Image",
            "Width" => Object::Integer(iw as i64),
            "Height" => Object::Integer(ih as i64),
            "ColorSpace" => "DeviceRGB",
            "BitsPerComponent" => Object::Integer(8),
            "Filter" => "FlateDecode",
        },
        compressed,
    );
    img_stream.set_content_compressed(false); // already compressed
    let img_id = host.add_object(img_stream);

    // Content stream: draw the image at page-fill size.
    let content = format!("q {} 0 0 {} 0 0 cm /Im0 Do Q", pw, ph);
    let content_id = host.add_object(lopdf::Stream::new(
        dictionary! {},
        content.into_bytes(),
    ));
    let resources = host.add_object(dictionary! {
        "XObject" => dictionary! { "Im0" => img_id },
        "ProcSet" => vec!["PDF".into(), "ImageC".into()],
    });
    // Parent will be patched up by the splice code below; for now use a
    // placeholder ID that the caller rewires.
    let pages_root_id = host.trailer.get(b"Root")?.as_reference()
        .and_then(|root| host.get_object(root)?.as_dict()?.get(b"Pages")?.as_reference())?;
    let page_id = host.add_object(dictionary! {
        "Type" => "Page",
        "Parent" => pages_root_id,
        "MediaBox" => vec![0.into(), 0.into(), Object::Real(pw), Object::Real(ph)],
        "Resources" => resources,
        "Contents" => content_id,
    });
    Ok(vec![page_id])
}

fn flate_compress(data: &[u8]) -> Vec<u8> {
    use flate2::write::ZlibEncoder;
    use flate2::Compression;
    use std::io::Write;
    let mut enc = ZlibEncoder::new(Vec::new(), Compression::default());
    enc.write_all(data).unwrap();
    enc.finish().unwrap()
}
```

```typescript
// src/lib/pagesHistory.ts
import { writable, type Writable } from "svelte/store";

export type PageOp =
  | { kind: "delete"; at: number }
  | { kind: "duplicate"; from: number; to: number }
  | { kind: "rotate"; at: number; degrees: 90 | 180 | 270 }
  | { kind: "reorder"; order: number[] }
  | { kind: "insert_blank"; at: number; count: number; width: number; height: number }
  | { kind: "insert_pdf"; at: number; path: string }
  | { kind: "insert_image"; at: number; path: string; dpi: number };

export interface HistorySnapshot {
  ops: PageOp[];
  cursor: number; // index of next un-applied slot; 0 == empty
  canUndo: boolean;
  canRedo: boolean;
}

export interface PagesHistory {
  store: Writable<HistorySnapshot>;
  push(op: PageOp): void;
  undo(): PageOp | null;
  redo(): PageOp | null;
  snapshot(): HistorySnapshot;
  clear(): void;
}

const DEFAULT_CAP = 50;

export function createPagesHistory(
  { cap = DEFAULT_CAP }: { cap?: number } = {},
): PagesHistory {
  let ops: PageOp[] = [];
  let cursor = 0;
  const store = writable<HistorySnapshot>(snapshot());

  function snapshot(): HistorySnapshot {
    return {
      ops: ops.slice(),
      cursor,
      canUndo: cursor > 0,
      canRedo: cursor < ops.length,
    };
  }

  function commit() {
    store.set(snapshot());
  }

  return {
    store,
    push(op) {
      // Truncate redo branch.
      ops = ops.slice(0, cursor);
      ops.push(op);
      // FIFO evict if over cap.
      if (ops.length > cap) {
        ops = ops.slice(ops.length - cap);
      }
      cursor = ops.length;
      commit();
    },
    undo() {
      if (cursor === 0) return null;
      cursor -= 1;
      commit();
      return ops[cursor];
    },
    redo() {
      if (cursor >= ops.length) return null;
      const op = ops[cursor];
      cursor += 1;
      commit();
      return op;
    },
    snapshot,
    clear() {
      ops = [];
      cursor = 0;
      commit();
    },
  };
}
```

```rust
use crate::pdf::{insert, pages, PdfError};
use std::path::{Path, PathBuf};

/// Apply a sequence of PageOps in order, writing the result atomically
/// to `output`. Each op is replayed against the working file via a
/// temporary cascade so the host PDF on disk is touched exactly once
/// (via `atomic_save`).
pub fn apply_ops(input: &Path, ops: &[PageOp], output: &Path) -> Result<(), PdfError> {
    if ops.is_empty() {
        let bytes = std::fs::read(input)?;
        return crate::pdf::atomic_save(output, &bytes);
    }
    // Cascade through a temp file per op, final cascade lands atomically
    // on `output`.
    let tmp_dir = std::env::temp_dir().join(format!("slab-pageops-{}", std::process::id()));
    std::fs::create_dir_all(&tmp_dir)?;
    let mut cur: PathBuf = input.to_path_buf();
    let mut idx = 0u32;
    for op in ops {
        let next = tmp_dir.join(format!("step-{idx}.pdf"));
        apply_one(&cur, op, &next)?;
        cur = next;
        idx += 1;
    }
    let bytes = std::fs::read(&cur)?;
    crate::pdf::atomic_save(output, &bytes)?;
    let _ = std::fs::remove_dir_all(&tmp_dir);
    Ok(())
}

fn apply_one(input: &Path, op: &PageOp, output: &Path) -> Result<(), PdfError> {
    match op {
        PageOp::Delete { at } => {
            pages::delete_pages(input, &[*at], output).map(|_| ())
        }
        PageOp::Duplicate { from, to: _ } => {
            pages::duplicate_pages(input, &[*from], output).map(|_| ())
        }
        PageOp::Rotate { at, degrees } => {
            let rot = pages::Rotation::from_int(*degrees)?;
            pages::rotate_pages_permanent(input, &[*at], rot, output).map(|_| ())
        }
        PageOp::Reorder { order } => {
            pages::reorder_pages(input, order, output)
        }
        PageOp::InsertBlank { at, count, width, height } => {
            insert::insert(input, output, insert::InsertOpts {
                at: *at,
                source: insert::InsertSource::Blank {
                    count: *count, width: *width, height: *height,
                },
            }).map(|_| ())
        }
        PageOp::InsertPdf { at, path } => {
            insert::insert(input, output, insert::InsertOpts {
                at: *at,
                source: insert::InsertSource::Pdf { path: path.clone() },
            }).map(|_| ())
        }
        PageOp::InsertImage { at, path } => {
            insert::insert(input, output, insert::InsertOpts {
                at: *at,
                source: insert::InsertSource::Image {
                    path: path.clone(), dpi: 72.0,
                },
            }).map(|_| ())
        }
    }
}
```
