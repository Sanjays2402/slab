> **For Hermes:** Use `subagent-driven-development` to implement this plan task-by-task. Each slice = one delegated subagent. Two-stage review (spec compliance, then code quality) between slices.

**Goal:** Add four shape annotation primitives (rectangle, ellipse, arrow, straight line) as the fourth annotation family — joining Highlight (v2.0.6), Note (v2.0.6), and Ink (v2.0.7). Closes the "Acrobat-tier markup toolkit" gap for v1.0.0 launch reviewers.

**Architecture:** Pure additive variant extension of the v2.0.6/v2.0.7 stack. Existing `Annotation` tagged-union in Rust gains four new variants (`Rect`, `Ellipse`, `Arrow`, `Line`); existing `annotations.ts` sidecar JSON gains four sibling `kind: "rect" | "ellipse" | "arrow" | "line"` records; `DrawLayer.svelte` from v2.0.7 gains a shape sub-mode union; `ActionId` enum gains `AnnotShape`. Zero new persistence layers, zero new Tauri commands, zero new third-party deps. Reuses v2.0.7's `pagerendered` overlay hook, v2.0.6's per-doc 200/3000-cap eviction policy, v2.0.7's CSS-px ↔ PDF-pt converter, and v2.0.7's RDP simplifier's bounding-box helper.

```typescript
  kind: "rect";
  page_index: number;
  rect: PdfRect;
  color: string;     // "#rrggbb"
  width: number;     // PDF-pt, clamped 0.5..12

  created_at: string;
}

export interface EllipseAnnotation { /* same shape, kind: "ellipse" */ }

export interface ArrowAnnotation {
  id: string;
  kind: "arrow";
  page_index: number;
  p0: PdfPoint;
  p1: PdfPoint;
  color: string;
  width: number;
  head: ArrowHead;
  opacity: number;
  created_at: string;
}

export interface LineAnnotation {
  id: string;
  kind: "line";
  page_index: number;
  p0: PdfPoint;
  p1: PdfPoint;
  color: string;
  width: number;
  dashed: boolean;
  opacity: number;
  created_at: string;

// Extend the existing Annotation union (v2.0.6 + v2.0.7):
export type Annotation =
  | HighlightAnnotation
  | NoteAnnotation
  | InkAnnotation
  | RectAnnotation
  | EllipseAnnotation
  | ArrowAnnotation
  | LineAnnotation;

const HEX6 = /^#[0-9a-fA-F]{6}$/;

function clampWidth(w: number): number {
  if (!Number.isFinite(w)) throw new TypeError("width must be finite");
  return Math.max(0.5, Math.min(12, w));
}

function checkColor(c: string) {
  if (!HEX6.test(c)) throw new TypeError(`color must be 6-digit hex, got ${c}`);
}

export function addRectAnnotation(
  path: string,
  pageIndex: number,
  rect: PdfRect,
  color: string,
  width = 2,
  opacity = 1,
): string {
  if (rect.w <= 0 || rect.h <= 0) throw new RangeError("rect must have positive w/h");
  checkColor(color);
  const rec: RectAnnotation = {
    id: `rect-${crypto.randomUUID()}`,
    kind: "rect",
    page_index: pageIndex,
    rect,
    color,
    width: clampWidth(width),
    opacity: Math.max(0, Math.min(1, opacity)),
    created_at: new Date().toISOString(),
  };
  appendRecord(path, rec);
  return rec.id;
}

// addArrowAnnotation: reject p0==p1; ArrowHead whitelist check
// addLineAnnotation: dashed defaults to false
```

```rust
#[derive(Serialize, Deserialize, Debug, Clone, Copy, PartialEq)]

fn hex_to_rgb(hex: &str) -> [f32; 3] {
    let r = u8::from_str_radix(&hex[1..3], 16).unwrap_or(0) as f32 / 255.0;
    let g = u8::from_str_radix(&hex[3..5], 16).unwrap_or(0) as f32 / 255.0;
    let b = u8::from_str_radix(&hex[5..7], 16).unwrap_or(0) as f32 / 255.0;
    [r, g, b]
}

fn rect_to_pdf_array(r: &PdfRect) -> Vec<Object> {
    vec![
        Object::Real(r.x),
        Object::Real(r.y),
        Object::Real(r.x + r.w),
        Object::Real(r.y + r.h),
    ]
}

// New match arms in build_annotation_dict():
Annotation::Rect { rect, color, width, opacity, .. } => {
    let mut dict = Dictionary::new();
    dict.set("Type", Object::Name(b"Annot".to_vec()));
    dict.set("Subtype", Object::Name(b"Square".to_vec()));
    dict.set("Rect", Object::Array(rect_to_pdf_array(rect)));
    let rgb = hex_to_rgb(color);

    dict.set("CA", Object::Real(*opacity));
    dict.set("F", Object::Integer(4));
```

```svelte
<!-- src/lib/panels/DrawLayer.svelte -->
<script lang="ts">
  export let mode:
    | "off"
    | "highlight"
    | "note"
    | "ink"
    | "rect"
    | "ellipse"
    | "arrow"
    | "line" = "off";

  let dragStart: PdfPoint | null = null;
  let dragEnd: PdfPoint | null = null;
  let shiftDown = false;

  function onPointerDown(e: PointerEvent) {
    if (!isShapeMode(mode)) return passToInk(e);
    e.preventDefault();
    dragStart = cssToPdf(e.offsetX, e.offsetY);
    dragEnd = dragStart;
    (e.target as Element).setPointerCapture(e.pointerId);
  }

  function onPointerMove(e: PointerEvent) {
    if (dragStart === null) return passToInk(e);
    dragEnd = cssToPdf(e.offsetX, e.offsetY);
    if (shiftDown) dragEnd = constrainAspect(mode, dragStart, dragEnd);
  }

  function onPointerUp(e: PointerEvent) {
    if (dragStart === null) return passToInk(e);
    const a = dragStart, b = dragEnd!;
    dragStart = null;
    dragEnd = null;
    if (!isMeaningfulShape(a, b)) return; // < 3px → click, not drag

    switch (mode) {
      case "rect":
        addRectAnnotation(path, pageIndex, normRect(a, b), color, width, opacity);
        break;
      case "ellipse":
        addEllipseAnnotation(path, pageIndex, normRect(a, b), color, width, opacity);
        break;
      case "arrow":
        addArrowAnnotation(path, pageIndex, a, b, color, width, arrowHead, opacity);
        break;
      case "line":
        addLineAnnotation(path, pageIndex, a, b, color, width, dashed, opacity);
        break;
    }
    flashCommitPulse(); // shared with v2.0.7 ink-trail-shimmer
  }

  function constrainAspect(m: typeof mode, a: PdfPoint, b: PdfPoint): PdfPoint {
    if (m === "rect" || m === "ellipse") {
      const side = Math.min(Math.abs(b.x - a.x), Math.abs(b.y - a.y));
      return {
        x: a.x + Math.sign(b.x - a.x) * side,
        y: a.y + Math.sign(b.y - a.y) * side,
      };
    }
    // For arrow & line: snap to nearest 45° increment
    const dx = b.x - a.x, dy = b.y - a.y;
    const angle = Math.atan2(dy, dx);
    const snap = Math.round(angle / (Math.PI / 4)) * (Math.PI / 4);
    const len = Math.hypot(dx, dy);
    return { x: a.x + len * Math.cos(snap), y: a.y + len * Math.sin(snap) };
  }
</script>
```

```svelte
      ctx.strokeStyle = r.color;
      ctx.lineWidth = r.width * vp.scale;
      switch (r.kind) {
        case "rect":   paintRect(ctx, vp, r); break;
        case "ellipse":paintEllipse(ctx, vp, r); break;
        case "arrow":  paintArrow(ctx, vp, r); break;
        case "line":   paintLine(ctx, vp, r); break;
      }
      ctx.restore();
    }
  }
</script>
```
