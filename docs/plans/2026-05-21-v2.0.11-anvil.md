# v2.0.11 "Anvil" ⚒️ — Multi-Select + Rotate + Align Implementation Plan

> **For Hermes:** Use `subagent-driven-development` to implement this plan task-by-task. Each slice = one delegated subagent. Two-stage review (spec compliance, then code quality) between slices.

**Goal:** Take v2.0.10's single-annotation editor and turn it into a multi-annotation editor — drag a marquee to select many, rotate any of the eight kinds via a real rotate handle, align/distribute selected annotations like Figma. Redeems the two open-question deferrals stacked into v2.0.10's plan ("Group/lasso multi-select → v2.0.11" + "Rotate handle → v2.0.11"), plus one new wow (alignment tools) that pushes Slab past every paid PDF editor on raw markup power.

**Architecture:** Extend `selection.ts` from `SelectionRef | null` to `SelectionSet` — a `Set<id>` per active path with a derived `primary` (last clicked) for handle anchoring. New `MarqueeLayer.svelte` (sibling of v2.0.10's `SelectionLayer.svelte`) listens to pointer-down on empty page area, paints an accent-tinted dashed rect via SVG, and on pointer-up converts the rect to a hit-test against `listAnnotations(activePath)` for the current page. Add a `rotation: number` (degrees, default 0) field to every annotation variant — **breaking sidecar JSON change**, so v2.0.11 ships a migration: sidecar files at `v: 1` get rewritten to `v: 2` on first load with `rotation: 0` injected. Backwards reads OK; forwards reads warn-and-coerce. Rust `Annotation` enum gains the same field; `build_annotation_dict()` emits PDF `/Rotate` integer rounded to nearest 90° (PDF spec quirk — only multiples allowed in `/Rect` form; for non-multiples we bake the rotation into the on-render quad math, which is already how Acrobat does it). New `align.ts` pure-TS module implements 6 alignment ops (left/center-x/right/top/center-y/bottom) + 2 distribute ops (horizontal/vertical) over a `SelectionSet`. Every multi-select mutation routes through v2.0.10's `history.apply(cmd)` as one composite command (one `Cmd+Z` undoes the whole alignment, not 12 individual moves). **One new persistence-format version (sidecar v1→v2)**, **zero new Tauri commands**, **zero new third-party deps**.

**Tech Stack:** Svelte 4 (frontend), Rust 1.78 / Tauri 2 (backend — `Annotation` enum gets a flat `rotation` field via `#[serde(default)]` for backwards compat; `build_annotation_dict` updated to emit `/Rotate`), pdfjs-dist 4 `eventBus.pagerendered` (existing), pure-TS marquee hit-test (no `IntersectionObserver` — direct rect-intersection math, < 200 annotations is trivial), CSS `transform: rotate()` for live preview during rotation gesture (zero pdfjs re-rasters).

## Slice 1 — Sidecar v1→v2 migration + `rotation` field

**Objective:** Add a `rotation: number` (degrees, default 0) field to every annotation kind in both TS and Rust, with a one-way migration of existing v1 sidecar files. Backwards-compatible reads, forward-emit at v2.

**Files:**
- Modify: `src/lib/annotations.ts` (~110 LOC) — bump `SIDECAR_VERSION` from `1` to `2`. Add `rotation: number` to every variant interface (default 0). - Modify: `src/lib/annotations.test.ts` (~50 LOC) — 6 new vitest cases covering load-v1-write-v2, load-v2-unchanged, load-v3-coerce, rotation default 0, rotation clamping, mixed-version directory.

**Step 2: Write the migration module.**

```ts
// src/lib/annotations-migrate.ts
import type { SidecarFile, Annotation } from './annotations';

export function migrateV1ToV2(raw: unknown): SidecarFile | null {
  if (typeof raw !== 'object' || !raw) return null;
  const r = raw as any;
  if (r.v === 2) return r as SidecarFile;
  if (r.v !== 1) return null;
  if (!Array.isArray(r.annotations)) return null;
  const annotations: Annotation[] = r.annotations.map((a: any) => ({
    ...a,
    rotation: typeof a.rotation === 'number' ? clamp(a.rotation) : 0,
  }));
  return { v: 2, annotations };
}

function clamp(n: number): number {
  if (!Number.isFinite(n)) return 0;
  return Math.max(-360, Math.min(360, n));
}
```

```ts
  if (!raw) return { v: SIDECAR_VERSION, annotations: [] };
```

```ts
// src/lib/selection.ts
import { writable, derived, get } from 'svelte/store';

export interface SelectionState {
  path: string | null;
  ids: Set<string>;
  primary: string | null;
}

const _selection = writable<SelectionState>({ path: null, ids: new Set(), primary: null });

export const selection = {
  subscribe: _selection.subscribe,
  select(path: string, id: string) {
    _selection.set({ path, ids: new Set([id]), primary: id });
  },
  extend(id: string) {
    _selection.update(s => {
      if (!s.path) return s;
      const next = new Set(s.ids);
      next.add(id);
      return { ...s, ids: next, primary: id };
    });
  },
  toggle(id: string) {
    _selection.update(s => {
      const next = new Set(s.ids);
      if (next.has(id)) next.delete(id); else next.add(id);
      return { ...s, ids: next, primary: next.has(id) ? id : (next.size ? [...next][next.size-1] : null) };
    });
  },
  selectMany(path: string, ids: string[]) {
    _selection.set({ path, ids: new Set(ids), primary: ids[ids.length-1] ?? null });
  },
  clear() { _selection.set({ path: null, ids: new Set(), primary: null }); },
  isSelected(id: string): boolean { return get(_selection).ids.has(id); },
  get(): SelectionState { return get(_selection); },
};

export const selectionSize = derived(_selection, $s => $s.ids.size);
```

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  import { selection } from '$lib/selection';
  import { listAnnotations, type Annotation } from '$lib/annotations';
  import { boundingBox } from '$lib/geometry';

  export let path: string;
  export let page: number;
  export let viewport: any;  // pdfjs viewport

  let marquee: { x0: number; y0: number; x1: number; y1: number } | null = null;
  let dragging = false;

  function onPointerDown(ev: PointerEvent) {
    if ((ev.target as HTMLElement).closest('[data-annot-id], [data-selection-handle]')) return;
    if (ev.button !== 0) return;
    const r = (ev.currentTarget as HTMLElement).getBoundingClientRect();
    const x = ev.clientX - r.left, y = ev.clientY - r.top;
    marquee = { x0: x, y0: y, x1: x, y1: y };
    dragging = true;
    if (!ev.shiftKey) selection.clear();
    (ev.currentTarget as HTMLElement).setPointerCapture(ev.pointerId);
    ev.preventDefault();
  }

  function onPointerMove(ev: PointerEvent) {
    if (!dragging || !marquee) return;
    const r = (ev.currentTarget as HTMLElement).getBoundingClientRect();
    marquee = { ...marquee, x1: ev.clientX - r.left, y1: ev.clientY - r.top };
  }

  function onPointerUp(ev: PointerEvent) {
    if (!dragging || !marquee) return;
    dragging = false;
    const { x0, y0, x1, y1 } = marquee;
    const rect = { x: Math.min(x0,x1), y: Math.min(y0,y1), w: Math.abs(x1-x0), h: Math.abs(y1-y0) };
    if (rect.w < 4 && rect.h < 4) { marquee = null; return; } // click, not drag
    const hits = listAnnotations(path)
      .filter(a => a.page === page)
      .filter(a => intersects(boundingBox(a, viewport), rect))
      .map(a => a.id);
    if (ev.shiftKey) {
      hits.forEach(id => selection.extend(id));
    } else {
      selection.selectMany(path, hits);
    }
    marquee = null;
  }

  function intersects(a: any, b: any): boolean {
    return !(a.x + a.w < b.x || b.x + b.w < a.x || a.y + a.h < b.y || b.y + b.h < a.y);
  }
</script>

<div class="marquee-layer"
     on:pointerdown={onPointerDown}
     on:pointermove={onPointerMove}
     on:pointerup={onPointerUp}>
  {#if marquee}
    {@const m = marquee}
    <svg class="marquee-svg">
      <rect x={Math.min(m.x0,m.x1)} y={Math.min(m.y0,m.y1)}
            width={Math.abs(m.x1-m.x0)} height={Math.abs(m.y1-m.y0)}
            fill="var(--accent-12)" stroke="var(--accent)" stroke-width="1.5"
            stroke-dasharray="4 3" />
    </svg>
  {/if}
</div>

<style>
  .marquee-layer { position: absolute; inset: 0; pointer-events: auto; z-index: 10; }
  .marquee-svg { position: absolute; inset: 0; pointer-events: none; }
  rect { animation: marquee-pulse 1.2s linear infinite; }
  @keyframes marquee-pulse { from { stroke-dashoffset: 0; } to { stroke-dashoffset: -14; } }
  @media (prefers-reduced-motion: reduce) { rect { animation: none; } }
</style>
```

**Step 1: Geometry helpers.**

```ts
// src/lib/geometry/rotation.ts
export function angleBetween(cx: number, cy: number, px: number, py: number): number {
  return (Math.atan2(py - cy, px - cx) * 180) / Math.PI;
}

export function snapToIncrement(deg: number, step: number): number {
  return Math.round(deg / step) * step;
}

export function applyRotation<T extends { rotation: number }>(annot: T, deltaDeg: number): T {
  const next = ((annot.rotation + deltaDeg) % 360 + 360) % 360;
  return { ...annot, rotation: next };
}
```

**Step 2: Wire the handle in `SelectionLayer.svelte`.**

```svelte
<script lang="ts">
  // … existing imports
  import { angleBetween, snapToIncrement, applyRotation } from '$lib/geometry/rotation';
  import { history, composeCommands } from '$lib/history';
  import { updateAnnotation, listAnnotations } from '$lib/annotations';

  let rotating = false;
  let startAngle = 0;
  let liveDelta = 0;
  let chipPos = { x: 0, y: 0 };

  function onRotateStart(ev: PointerEvent) {
    const sel = selection.get();
    if (!sel.path || sel.ids.size === 0) return;
    const center = unionCenter(sel);
    startAngle = angleBetween(center.x, center.y, ev.clientX, ev.clientY);
    rotating = true;
    (ev.currentTarget as HTMLElement).setPointerCapture(ev.pointerId);
    ev.preventDefault();
  }

  function onRotateMove(ev: PointerEvent) {
    if (!rotating) return;
    const sel = selection.get();
    const center = unionCenter(sel);
    let delta = angleBetween(center.x, center.y, ev.clientX, ev.clientY) - startAngle;
    if (ev.shiftKey) delta = snapToIncrement(delta, 15);
    liveDelta = delta;
    chipPos = { x: ev.clientX + 12, y: ev.clientY - 24 };
    // apply CSS transform to every selected annotation node
    for (const id of sel.ids) {
      const node = document.querySelector(`[data-annot-id="${id}"]`) as HTMLElement | null;
      if (node) node.style.transform = `rotate(${delta}deg)`;
    }
  }

  function onRotateEnd(ev: PointerEvent) {
    if (!rotating) return;
    rotating = false;
    const sel = selection.get();
    if (!sel.path || liveDelta === 0) return;
    // clear inline transforms (committed values take over)
    for (const id of sel.ids) {
      const node = document.querySelector(`[data-annot-id="${id}"]`) as HTMLElement | null;
      if (node) node.style.transform = '';
    }
    const cmds = [...sel.ids].map(id => {
      const before = getAnnotation(sel.path!, id)!;
      const after = applyRotation(before, liveDelta);
      return {
        label: `Rotate ${before.kind}`,
        do: () => updateAnnotation(sel.path!, id, after),
        undo: () => updateAnnotation(sel.path!, id, before),
      };
    });
    history.apply(sel.path, composeCommands(`Rotate ${sel.ids.size} annotations`, cmds));
    liveDelta = 0;
  }
</script>

{#if selection.get().ids.size > 0 && !rotating}
```

```ts
// src/lib/align.ts
import type { Annotation } from './annotations';
import { boundingBox } from './geometry';

interface AlignResult { id: string; before: Annotation; after: Annotation; }

export function alignLeft(annots: Annotation[], viewport: any): AlignResult[] {
  if (annots.length < 2) return [];
  const targetX = Math.min(...annots.map(a => boundingBox(a, viewport).x));
  return annots.map(a => ({
    id: a.id,
    before: a,
    after: translate(a, targetX - boundingBox(a, viewport).x, 0),
  }));
}

export function distributeHorizontal(annots: Annotation[], viewport: any): AlignResult[] {
  if (annots.length < 3) return [];
  const sorted = [...annots].sort((a, b) =>
    boundingBox(a, viewport).x - boundingBox(b, viewport).x);
  const minX = boundingBox(sorted[0], viewport).x;
  const maxX = boundingBox(sorted.at(-1)!, viewport).x;
  const step = (maxX - minX) / (sorted.length - 1);
  return sorted.map((a, i) => {
    const target = minX + i * step;
    const dx = target - boundingBox(a, viewport).x;
    return { id: a.id, before: a, after: translate(a, dx, 0) };
  });
}
```

```svelte
  async function runOp(op: (annots: any[], v: any) => any[], label: string) {
    const sel = selection.get();
    if (!sel.path || sel.ids.size < 2) return;
    const annots = [...sel.ids].map(id => getAnnotation(sel.path!, id)).filter(Boolean) as any[];
    const results = op(annots, viewport);
    if (results.length === 0) return;
    // animate
    for (const { id } of results) {
      const node = document.querySelector(`[data-annot-id="${id}"]`) as HTMLElement | null;
      if (node) node.setAttribute('data-aligning', 'true');
    }
    history.apply(sel.path, composeCommands(label, results.map(r => ({
      label,
      do: () => updateAnnotation(sel.path!, r.id, r.after),
      undo: () => updateAnnotation(sel.path!, r.id, r.before),
    }))));
    await tick();
    setTimeout(() => {
      for (const { id } of results) {
        const node = document.querySelector(`[data-annot-id="${id}"]`) as HTMLElement | null;
        if (node) node.removeAttribute('data-aligning');

    }, 280);
```
